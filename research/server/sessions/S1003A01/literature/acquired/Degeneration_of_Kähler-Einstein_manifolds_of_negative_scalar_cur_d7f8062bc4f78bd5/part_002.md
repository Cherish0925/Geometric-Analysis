\item[(1)] Suppose $\widetilde{X} \cap U_{\beta} \neq \phi$. We then assume that $\widetilde{X}=\{x=0\}$ in $U_{\beta}$ and so $\left\{z_{k}=0\right\}$ corresponds to one of $E_{i}, F_{j}$ and $\widetilde{X}^{\prime}$. Let

be the flat conical Kähler metric on $U_{\beta}$. For each suitable sufficiently small $\epsilon_{4}$, there exists $C_{1}=C_{1}>0$,

where $\epsilon=\left(\epsilon_{1}, \epsilon_{2}, \epsilon_{3}, \epsilon_{4}\right)$. Since
$$d t^{\prime} \wedge d \bar{t}^{\prime}=\left(z_{1} \ldots z_{m} d x+\sum_{k=1}^{m} \frac{x z_{1} \ldots z_{m}}{z_{k}} d z_{k}\right) \wedge\left(\bar{z}_{1} \ldots \bar{z}_{m} d \bar{x}+\sum_{k=1}^{m} \frac{\overline{x z_{1}} \ldots \bar{z}_{m}}{\bar{z}_{k}} d \bar{z}_{k}\right),$$

Since we can obtain estimates for $\varphi_{t^{\prime}}$ near any component of $\mathcal{X}_{0}^{\prime}$, Lemma 4.2 immediately implies the following local estimates for $\varphi_{t}$ on $\mathcal{X}$.

$$\omega_{t^{\prime}} \geq C_{\epsilon}\left(\left.F_{\widetilde{X}}\right|_{\mathcal{X}_{t^{\prime}}^{\prime}}\right)^{\epsilon} \chi_{t^{\prime}} .$$

Proof. Let
$$H=\log \left(\left.F_{\widetilde{X}}\right|_{\mathcal{X}_{t^{\prime}}^{\prime}}\right)^{\epsilon} \operatorname{tr}_{\omega_{t^{\prime}}}\left(\chi_{t^{\prime}}\right)-2 A \varphi_{t^{\prime}}$$

for sufficiently large $A>0$ and some uniform constant $C_{1}>0$ dependent on $A$. Applying the maximum principle and the estimate $\varphi_{t^{\prime}}$ from Lemma 4.2, there exists $C_{2}>0$ such that for all $t^{\prime} \in\left(B^{\prime}\right)^{*}$,
$$\sup _{\mathcal{X}_{t^{\prime}}^{\prime}} H \leq C_{2} .$$
The lemma follows immediately from the uniform upper bound for $\varphi_{t^{\prime}}$ in Corollary 4.1. \(\square\)

The following lemma establishes the uniform non-collapsing condition for the Kähler-Einstein manifolds $\left(\mathcal{X}_{t}, g_{t}\right)$, for all $t \in B^{*}$.

Lemma 4.6. For any nonsingular point $p_{0} \in \mathcal{X}_{0}$, we can pick a smooth section $p(t)$ : $B \rightarrow \mathcal{X}$ such that $p(0)=p_{0}$. Then there exists $c>0$ such that for all $t \in B^{*}$,
$$\operatorname{Vol}_{g_{t}}\left(B_{g_{t}}(p(t), 1)\right) \geq c,$$
where $B_{g_{t}}(p(t), 1)$ is the unit geodesic ball centered at $p(t)$ in $\left(\mathcal{X}_{t}, g_{t}\right)$.
Proof. Since $p_{0}$ is a nonsingular point of $\mathcal{X}_{0}$, there exists $r>0$ such that the geodesic ball $B_{\chi_{0}}\left(p_{0}, r\right)$ in $\left(\mathcal{X}_{0}, \chi_{0}\right)$ completely lies in the nonsingular part of $\mathcal{X}_{0}$. Then there exists $c>0$ such that for all $t \in B$,
$$\operatorname{Vol}_{\chi_{t}}\left(B_{\chi_{t}}(p(t), r)\right)>c .$$

By Lemma 4.3, there exists $C>0$ such that for all $t \in B^{*}$,
$$C^{-1} \chi_{t} \leq \omega_{t} \leq C \chi_{t}$$
on $B_{\chi_{t}}(p(t), r)$ in $\left(\mathcal{X}_{t}, \chi_{t}\right)$. This implies that
$$B_{\chi_{t}}(p(t), r) \subset B_{g_{t}}\left(p(t), C^{1 / 2} r\right)$$
and so
$$\operatorname{Vol}_{g_{t}}\left(B_{g_{t}}\left(p(t), C^{1 / 2} r\right) \geq C^{-n} \operatorname{Vol}_{\chi_{t}}\left(B_{\chi_{t}}(p(t), r)\right) \geq C^{-n} c .\right.$$
The lemma then immediately follows by the volume comparison theorem. \(\square\)

The proof of Lemma 4.6 also shows that for any given nonsingular point $p_{0}$ in $\mathcal{X}_{0}$, the non-collapsing condition holds uniformly for all points near $p_{0}$.

\section*{5. Gromov-Hausdorff convergence and partial $C^{0}$-estimates}

Let $\pi: \mathcal{X} \rightarrow B$ be a stable degeneration of smooth canonical models as considered in Section 4. Suppose
$$\mathcal{X}_{0}=\bigcup_{\alpha=1}^{\mathcal{A}} X_{\alpha},$$
where each $X_{\alpha}$ is an irreducible component of $\mathcal{X}_{0}$. We pick any $\mathcal{A}$-tuple of nonsingular points
$$\left(p_{0}^{1}, p_{0}^{2}, \ldots, p_{0}^{\mathcal{A}}\right), p_{0}^{\alpha} \in X_{\alpha} \cap \mathcal{R}_{\mathcal{X}_{0}}, \alpha=1, \ldots, \mathcal{A} .$$

Let $\left(p_{t_{j}}^{1}, p_{t_{j}}^{2}, \ldots, p_{t_{j}}^{\mathcal{A}}\right)$ a sequence of $\mathcal{A}$-tuples of points $\in \mathcal{X}_{t_{j}}$ with $t_{j} \rightarrow 0$ such that
$$\left(p_{t_{j}}^{1}, p_{t_{j}}^{2}, \ldots, p_{t_{j}}^{\mathcal{A}}\right) \rightarrow\left(p_{0}^{1}, p_{0}^{2}, \ldots, p_{0}^{\mathcal{A}}\right)$$
with respect to the fixed reference metric $\chi$ on $\mathcal{X}$. Let $g_{t_{j}}$ be the corresponding Kähler-Einstein metric on $\mathcal{X}_{t_{j}}$. We would like to study the Riemannian geometric convergence of $\left(\mathcal{X}_{t_{j}}, g_{t_{j}}\right)$ as $t_{j} \rightarrow 0$.

Lemma 5.1. Let $g_{t}$ be the unique Kähler-Einstein metric on $\mathcal{X}_{t}$ for $t \in B^{*}$. After passing to a subsequence, $\left(\mathcal{X}_{t_{j}}, g_{t_{j}},\left(p_{t_{j}}^{1}, \ldots, p_{t_{j}}^{\mathcal{A}}\right)\right)$ converges in pointed Gromov-Hausdorff topology to a metric length space
$$\left(\mathbf{Y}, d_{\mathbf{Y}}\right)=\coprod_{\beta=1}^{\mathcal{B}}\left(Y_{\beta}, d_{\beta}\right)$$
as a disjoint union of metric length spaces $\left(Y_{\beta}, d_{\beta}\right)$, satisfying
\begin{itemize}
\item[(1)] $\mathbf{Y}=\mathcal{R}_{\mathbf{Y}} \cup \mathcal{S}_{\mathbf{Y}}$, where $\mathcal{R}_{\mathbf{Y}}$ and $\mathcal{S}_{\mathbf{Y}}$ are the regular and singular part of $\mathbf{Y} . \mathcal{R}_{\mathbf{Y}}$ is an open Kähler manifold and $\mathcal{S}_{\mathbf{Y}}$ is closed of Hausdorff dimension no greater than $2 n-4$.
\item[(2)] $g_{t_{j}}$ converge smoothly to a Kähler-Einstein metric $g_{K E}$ on $\mathcal{R}_{\mathbf{Y}}$. In particular, $g_{K E}$ coincides with the unique Kähler-Einstein current constructed in Theorem 1.1 on $\mathcal{X}_{0}$.
\item[(3)] $\mathcal{R}_{\mathcal{X}_{0}}$ is an open dense set in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ and
$$\mathcal{R}_{\mathcal{X}_{0}} \subset \mathcal{R}_{\mathbf{Y}} .$$
\end{itemize}
\begin{itemize}
\item[(4)] $\mathcal{B} \leq \mathcal{A}$ and
$$\operatorname{Vol}\left(\mathbf{Y}, d_{\mathbf{Y}}\right)=\sum_{\beta=1}^{\mathcal{B}} \operatorname{Vol}\left(Y_{\beta}, d_{\beta}\right)=\operatorname{Vol}\left(\mathcal{X}_{t}, g_{t}\right)$$
for all $t \in B^{*}$.
\end{itemize}

Proof. By Lemma 4.6, there exists $c>0$ and $>0$ such that for all $\alpha$ and $j$,
$$\operatorname{Vol}_{g_{t_{j}}}\left(B_{g_{t_{j}}}\left(p_{t_{j}}^{\alpha}, 1\right) \geq c .\right.$$
Then Cheeger-Colding-Tian theory [8] immediately implies that the pointed Gromov-Hausdorff convergence and (1), (2) hold.

Since $g_{t}$ converges smoothly on the nonsingular part $\mathcal{R}_{\mathcal{X}_{0}}, \mathcal{R}_{\mathcal{X}_{0}}$ must be contained in the regular part $\mathcal{R}_{\mathbf{Y}}$ of $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$.

The volume $\operatorname{Vol}_{g_{t}}\left(\mathcal{X}_{t}\right)=\left[K_{\mathcal{X}_{t}}\right]^{n}$ is a fixed constant for all $t \in B$, so by the volume convergence, we have
$$\operatorname{Vol}\left(\mathbf{Y}, d_{\mathbf{Y}}\right) \leq \operatorname{Vol}_{g_{t}}\left(\mathcal{X}_{t}\right)=\left[K_{\mathcal{X}_{t}}\right]^{n}$$
for all $t \in B^{*}$. On the other hand, $g_{K E}$ extends to the unique Kähler-Einstein current on $\mathcal{X}_{0}$ and so by Lemma 3.6 and Corollary 3.7,
$$\operatorname{Vol}\left(\mathbf{Y}, d_{\mathbf{Y}}\right) \geq \int_{\mathcal{R}_{\mathcal{X}_{0}}} d V_{g_{K E}}=\left[K_{\mathcal{X}_{0}}\right]^{n} .$$
This implies that $\mathcal{R}_{\mathcal{X}_{0}}$ must be dense in ( $\mathbf{Y}, d_{\mathbf{Y}}$ ).
Since $\mathcal{R}_{\mathcal{X}_{0}}$ has $\mathcal{A}$ disjoint components with total volume equal to the volume of $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$, there can be at most $\mathcal{A}$ disjoint components in ( $\mathbf{Y}, d_{\mathbf{Y}}$ ).

Therefore we have proved (4) and (5). \(\square\)

The following proposition can be proved by similar arguments in [45, 18] as local $L^{2}$ -estimates from Tian's proposal for the partial $C^{0}$-estimates. We let $h_{t}=\left(\left(\omega_{t}\right)^{n}\right)^{-1}=$ $\left(e^{\varphi_{t}} \Omega_{t}\right)^{-1}$ be the hermitian metric on $\mathcal{X}_{t}$ for $t \in B^{*}$, where $\omega_{t}$ is Kähler-Einstein form associated to the Kahelr-Einstein metric $g_{t}$ on $\mathcal{X}_{t}, \Omega_{t}$ and $\varphi_{t}$ are defined in Section 4.

Lemma 5.2. Let $p_{0}$ be a nonsingular point in $\mathcal{X}_{0}$ and $p: B \rightarrow \mathcal{X}$ be a smooth section with $p(0)=p_{0}$. For any $R>0$, there exists $K_{R}>0$ such that if $\sigma \in H^{0}\left(\mathcal{X}_{t}, m K_{\mathcal{X}_{t}}\right)$ for $m \geq 1$ with $t \in B^{*}$, then
$$\begin{gathered}
\|\sigma\|_{L^{\infty, \sharp}\left(B_{g_{t}}(p(t), R)\right)} \leq K_{R}\|\sigma\|_{L^{2, \sharp}\left(B_{g_{t}}(p(t), 2 R)\right)} \\
\|\nabla \sigma\|_{L^{\infty, \sharp}\left(B_{g(t)}(p(t), R)\right)} \leq K_{R}\|\sigma\|_{L^{2, \sharp}\left(B_{g_{t}}(p(t), 2 R)\right)},
\end{gathered}$$
where $B_{g_{t}}(p(t), R)$ is the geodesic ball centered at $p(t)$ with radius $R$ in $\left(\mathcal{X}_{t}, g_{t}\right)$, the $L^{2}$ -norms $\|\sigma\|_{L^{\infty, \sharp}}$ and $\|\nabla \sigma\|_{L^{\infty, \sharp}}$ are defined with respect to the rescaled hermitian metric $\left(h_{t}\right)^{m}$ and the rescaled Kähler metric $m g_{t}$.

Proof. For fix $R>0$, the Sobolev constant on $B_{g_{t}}(p(t), R)$ is uniformly bounded because of the Einstein condition and the uniform noncollapsing condition for unit balls centered at $p(t)$. The proof follows by well-known argument of Moser's iteration on balls of relative scales using cut-off functions (c.f [35]). \(\square\)

The following $L^{2}$-estimate is standard for Kähler-Einstein manifolds.
Lemma 5.3. For any integer $m \geq 2$, any $t \in B^{*}$ and any smooth $\left(m K_{\mathcal{X}_{t}}\right)$-valued $(0,1)$ - form $\tau$ satisfying $\bar{\partial} \tau=0$, there exists an $\left(m K_{\mathcal{X}_{t}}\right)$-valued section u such that $\bar{\partial} u=\tau$ and
$$\int_{\mathcal{X}_{t}}|u|_{\left(h_{t}\right)^{m}}^{2} d V_{g_{t}} \leq \frac{1}{2 \pi} \int_{\mathcal{X}_{t}}|\tau|_{\left(h_{t}\right)^{m}}^{2} d V_{g_{t}} .$$

The following lemma gives a construction for global pluricanonical section on the limiting metric space $\mathbf{Y}$.

Lemma 5.4. Suppose $t_{j} \in B^{*} \rightarrow 0$ and $\sigma_{t_{j}} \in H^{0}\left(\mathcal{X}_{t}, m K_{\mathcal{X}_{t}}\right)$ be a sequence of sections satisfying
$$\int_{\mathcal{X}_{t_{j}}}\left|\sigma_{t_{j}}\right|_{\left(h_{t_{j}}\right)^{m}}^{2} d V_{m g_{t_{j}}}=1$$
Then after passing to a subsequence, $\sigma_{t_{j}}$ converges to a holomorphic section $\sigma$ of $m K_{\mathbf{Y}}$. Furthermore, the $\left.\sigma\right|_{\mathcal{R}_{\mathcal{X}_{0}}}$ extends to a unique $\sigma^{\prime} \in H^{0}\left(\mathcal{X}_{0}, m K_{\mathcal{X}_{0}}\right)$ and $\sigma$ vanishes along $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$.

Proof. We choose any nonsingular point $p_{0}$ in $\mathcal{X}_{0}$ and a sequence $p_{t_{j}} \in \mathcal{X}_{t_{j}}$ such that $p_{t_{j}}$ converges to $p_{0}$ in $(\mathcal{X}, \chi)$. Then $p_{t_{j}}$ in $\left(\mathcal{X}_{t_{j}}, g_{t_{j}}\right)$ also converges in Gromov-Hausdorff distance to $p_{0}$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ due to the smooth convergence of $g_{t_{j}}$ to $g_{K E}$ on $\mathcal{R}_{\mathcal{X}_{0}}$. Since the $L^{2}$-norm of $\sigma_{t_{j}}$ with respect to $\left(h_{t_{j}}\right)^{m}$ and $m g_{t_{j}}$ is uniformly bounded, for any $R>0$, there exists $C_{R}>0$ such that for all $j$, we have
$$\sup _{B_{g_{t_{j}}}\left(p_{t_{j}}, R\right)}\left|\sigma_{t_{j}}\right| \leq C_{R}, \sup _{B_{g_{t_{j}}}\left(p_{t_{j}}, R\right)}\left|\nabla \sigma_{t_{j}}\right|_{g_{t_{j}}} \leq C_{R} .$$
After passing to sequence, $\sigma_{t_{j}}$ converges to a section $\sigma$ of $m K_{\mathbf{Y}}$ and it is a holomorphic on $\mathcal{R}_{\mathbf{Y}} \supset \mathcal{R}_{\mathcal{X}_{0}}$. It uniquely extends to the singular set of $\mathbf{Y}$ because of the gradient estimate and the geodesic convexity of $\mathcal{R}_{\mathbf{Y}}$.

Since $\sigma$ is a holomorphic section of $m K_{\mathcal{R}_{\mathcal{X}_{0}}}$, it can be uniquely extended to a pluricanonical section on the normal part of $\mathcal{X}_{0}$. Without loss of generality, we can assume that the adapted volume measure is given by $\Omega_{0}=\left(\sum_{j=0}^{N} \eta_{j} \wedge \overline{\eta_{j}}\right)^{1 / m}$, where $\left\{\eta_{j}\right\}$ gives a global projective embedding of $\mathcal{X}_{0}$. Let $\omega_{0}=\chi_{0}+\sqrt{-1} \partial \bar{\partial} \varphi_{0}$ be unique the Kähler-Einstein current on $\mathcal{X}_{0}$ constructed in Lemma 3.6 and Corollary 3.7, satisfying
$$\left(\chi_{0}+\sqrt{-1} \partial \bar{\partial} \varphi_{0}\right)^{n}=e^{\varphi_{0}} \Omega_{0} .$$
By the $L^{2}$-bound of $\sigma$ and the upper bound of $\varphi_{0}$ from Corollary 4.1, there exists $C>0$ such that
$$\int_{\mathcal{R}_{\mathcal{X}_{0}}} \frac{|\sigma|^{2}}{\left(\Omega_{0}\right)^{m}} \Omega_{0} \leq C \int_{\mathcal{R}_{\mathcal{X}_{0}}} \frac{|\sigma|^{2}}{\left(\Omega_{0}\right)^{m}} e^{-m \varphi_{0}} \Omega_{0}<\infty .$$
We can pullback the above formula on $\widetilde{\mathcal{X}_{0}}$, a log resolution of $\mathcal{X}_{0}$. The pullback of $\Omega_{0}$ on $\widetilde{\mathcal{X}_{0}}$ has poles of order 1 along the exceptional divisor over the non-log terminal locus $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$ of $\mathcal{X}_{0}$. Therefore $\sigma$ must vanish along $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$ so that the integral (5.21) is
finite. More precisely, for any point $x \in \operatorname{LCS}\left(\mathcal{X}_{0}\right)$, there exists an open neighborhood $U_{x}$ of $x$, such that $\frac{\sigma}{\eta_{i}}$ is bounded on $U_{x}$ for some $i$, and
$$\left.\frac{\sigma}{\eta_{i}}\right|_{U_{x} \cap \operatorname{LCS}\left(\mathcal{X}_{0}\right)}=0 .$$
As a consequence, $\sigma$ can be uniquely extended globally to $\sigma^{\prime}$ on $\mathcal{X}_{0}$. \(\square\)

The following is the local version of the partial $C^{0}$-estimate.
Lemma 5.5. Let $p_{0} \in \mathcal{R}_{\mathcal{X}_{0}}$ and $p: B \rightarrow \mathcal{X}$ be a smooth section with $p(0)=p_{0}$. For any $R>0$, there exist $m \in \mathbb{Z}^{+}$and $c>0$ such that for any $t \in B^{*}$ and $q \in B_{g_{t}}(p(t), R)$, there exists $\sigma_{t} \in H^{0}\left(\mathcal{X}_{0}, m K_{\mathcal{X}_{t}}\right)$ satisfying
$$\left|\sigma_{t}\right|_{\left(h_{t}\right)^{m}}^{2}(q) \geq c, \quad \int_{\mathcal{X}_{t}}\left|\sigma_{t}\right|_{\left(h_{t}\right)^{m}}^{2} d V_{m g_{t}}=1 .$$
Proof. The proof of the global partial $C^{0}$-estimate in [18] can be directly applied here in the local case with the estimates in Lemma 5.2 and Lemma 5.4 because the singular set of all iterated tangent cones of the Gromov-Hausdorff limit $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ is closed and has Hausdorff dimension less than $2 n-2$. \(\square\)

Let $h_{\mathbf{Y}}$ be the hermitian metric on $K_{\mathbf{Y}}$ as the extension of $\left(\left(\omega_{\mathbf{Y}}\right)^{n}\right)^{-1}$ from $\mathcal{R}_{\mathbf{Y}}$, where $\omega_{\mathbf{Y}}$ is the Kähler-Einstein form on $\mathcal{R}_{\mathbf{Y}}$. We now pass the partial $C^{0}$-estimate in Lemma 5.5 to the limiting space $\mathbf{Y}$.

Corollary 5.1. For any $p_{0} \in \mathcal{R}_{\mathbf{Y}}$ and any $R>0$, there exist $m \in \mathbb{Z}^{+}$and $c, C>0$ such that for any $q \in B_{d_{\mathbf{Y}}}(q, R)$, there exists $\sigma \in H^{0}\left(\mathbf{Y}, m K_{\mathbf{Y}}\right)$ satisfying
$$|\sigma|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2}(q) \geq c, \quad \int_{\mathbf{Y}}|\sigma|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2} d V_{d_{\mathbf{Y}}}=1 .$$

We remark that such $\sigma$ has uniformly bounded gradient estimate and it can also be extended to a global pluricanonical section on $\mathcal{X}_{0}$. There are many generalizations and variations of Lemma 5.5 and Corollary 5.1. For example, if $p$ is a regular point in Y , then there exist global pluricanonical sections $\sigma_{0}, \ldots, \sigma_{n}$ such that near $p, \sigma_{0}$ is nonzero and
$$\frac{\sigma_{1}}{\sigma_{0}}, \ldots, \frac{\sigma_{n}}{\sigma_{0}}$$
can be used as holomorphic local coordinates near $p$. The proof of Theorem 5.21 and Corollary 5.1 is also used to construct peak sections to separate distinct points on Y.

\section*{6. Distance estimates}

In this section, our goal is to estimate the distance from a singular point of $\mathcal{X}_{0}$ to a given nonsingular point of $\mathcal{X}_{0}$. We will establish a principle for geometric complex Monge-Ampère equations of our interest that boundedness of local potentials is equivalent to boundedness of distance.

First, we want to show that the regular part of the Gromov-Hausdorff limit coincides with the nonsingular part of $\mathcal{X}_{0}$.

Lemma 6.1. Let $\mathcal{R}_{\mathbf{Y}}$ be the regular part of the metric space $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ and $\mathcal{R}_{\mathcal{X}_{0}}$ be the nonsingular part of the projective variety of $\mathcal{X}_{0}$. Then
$$\mathcal{R}_{\mathfrak{Y}}=\mathcal{R}_{\mathcal{X}_{0}}$$
and they are biholomorphic to each other.
Proof. Obviously,
$$\mathcal{R}_{\mathfrak{Y}} \supset \mathcal{R}_{\mathcal{X}_{0}}$$
from Lemma 5.1 and it suffices to show the other direction.
We prove by contradiction. Suppose $q \in \mathcal{R}_{\mathbf{Y}} \backslash \mathcal{R}_{\mathcal{X}_{0}}$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$. Since $\mathcal{R}_{\mathcal{X}_{0}}$ is dense in $\mathbf{Y}$, there exist a sequence of points $q_{j} \in \mathcal{R}_{\mathcal{X}_{0}}$ such that $q_{j} \rightarrow q$ with respect to $d_{\mathbf{Y}}$ in $\mathbf{Y}$. We can assume after passing to a subsequence that $q_{j}$ converges to $q^{\prime} \in \mathcal{X}_{0}$ with respect to $\chi_{0}$ in $\mathcal{X}_{0}$.

Since $q$ is a regular point in ( $\mathbf{Y}, d_{\mathbf{Y}}$ ), there exists $r>0$ such that $B_{d_{\mathbf{Y}}}(q, r)$ is a smooth open domain in the Euclidean space $\mathbb{C}^{n}$, and $d_{\mathbf{Y}}$ induces a smooth Kähler metric on $B_{d_{\mathbf{Y}}}(q, r)$. Furthermore, there exist $t_{i} \rightarrow 0$ and $x_{i} \in \mathcal{X}_{t_{i}}$ such that, $B_{g_{t_{i}}}\left(x_{i}, r\right) \subset \mathcal{X}_{t_{i}}$ converges smoothly to $B_{d_{\mathbf{Y}}}(q, r)$. By the partial $C^{0}$-estimates for regular points, there exist global sections $\sigma_{0}, \sigma_{1}, \ldots, \sigma_{n}$ of $H^{0}\left(\mathbf{Y}, m K_{\mathbf{Y}}\right)$ for some sufficiently large $m \in \mathbb{Z}^{+}$such that after making $r$ sufficiently small, $\sigma_{0}$ does not vanish on $B_{d_{\mathbf{Y}}}(q, r)$ and
$$\frac{\sigma_{1}}{\sigma_{0}}, \ldots, \frac{\sigma_{n}}{\sigma_{0}}$$
are holomorphic local coordinates on $B_{d_{\mathbf{Y}}}(q, r)$. In particular, the $L^{2}$-norm of each $\sigma_{k}$ is bounded with respect to $h_{\mathbf{Y}}$ and $d_{\mathbf{Y}}$. Therefore we have a biholomrphism
$$F: B_{d_{\mathbf{Y}}}(q, r) \rightarrow \mathcal{V} \in \mathbb{C}^{n}$$
by letting $F=\left(\frac{\sigma_{1}}{\sigma_{0}}, \ldots, \frac{\sigma_{n}}{\sigma_{0}}\right)$.
Also each $\sigma_{k}$ can be uniquely extended to a pluricanonical section on $\mathcal{X}_{0}$, for $k=$ $0,1, \ldots n$. We write it as $\sigma_{k}^{\prime}$. As before, we can assume that the adapted volume measure $\Omega_{0}=\left(\sum_{l=0}^{N} \eta_{l} \wedge \overline{\eta_{l}}\right)^{1 / m}$, where $\left\{\eta_{l}\right\}_{l=0}^{N}$ gives a global projective embedding of $\mathcal{X}_{0}$. We will discuss in the following two cases.
\begin{itemize}
\item[(1)] Suppose $q^{\prime} \in \operatorname{LCS}\left(\mathcal{X}_{0}\right)$. The Kähler-Einstein volume form on $B_{d_{\mathbf{Y}}}(q, r)$ is given by
$$\left|\sigma_{0}\right|^{2 / m} e^{\varphi \mathcal{v}},$$
where $\varphi_{\mathcal{V}}$ is a smooth bounded plurisubharmonic function on $B_{d_{\mathbf{Y}}}(q, r)$. Therefore on $\mathcal{R}_{\mathcal{X}_{0}} \cap B_{d_{\mathbf{Y}}}(q, r)$, we have
$$\left|\sigma_{0}\right|^{2 / m}=\left(\sum_{l=0}^{N} \eta_{l} \wedge \overline{\eta_{l}}\right)^{1 / m} e^{\varphi_{0}-\varphi_{\mathcal{V}}},$$
where $\varphi_{0}$ satisfies the Monge-Ampère equation
$$\left(\chi_{0}+\sqrt{-1} \partial \bar{\partial} \varphi_{0}\right)^{n}=e^{\varphi_{0}} \Omega_{0} .$$
\end{itemize}
\begin{itemize}
\item[] Suppose $\eta_{0}$ generates $m K_{\mathcal{X}_{0}}$ at $q^{\prime}$. Then
$$f=\frac{\sigma_{0}^{\prime}}{\eta_{0}}$$
is a meromorphic function on $\mathcal{X}_{0}$, where $\sigma_{k}^{\prime}$ is the unique extension of $\sigma_{k}$ from $\mathcal{R}_{\mathcal{X}_{0}}$ to $\mathcal{X}_{0}$ for $k=0,1, \ldots, n$. $f$ is a holomorphic function in a neighborhood $U_{q^{\prime}}$ in $\mathcal{X}_{0}$ and $f\left(q^{\prime}\right)=0$ since $\sigma_{0}^{\prime}$ vanishes at $q^{\prime}$. We can replace $U_{q^{\prime}}$ by a smooth neighborhood $\widetilde{U_{q^{\prime}}}$ after a log resolution of $\mathcal{X}_{0}$. Then the pullback of $f$ on $U_{q^{\prime}}$ vanishes along a divisor $E$ (containing the exceptional divisor over $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$ ) and so it must vanish to order of at least 1. On the other hand, after pulling back $\varphi_{0}$ to $\widetilde{U_{q^{\prime}}}$, for any $\epsilon>0$ there exists $C_{\epsilon}>0$ such that at each pre-image of $q_{j}$ in $\widetilde{U_{q^{\prime}}}$,
$$|f|^{2} \geq e^{\varphi_{0}-\varphi_{\mathcal{V}}} \geq C_{\epsilon}|f|^{2 \epsilon}$$
because $\varphi_{0}$ is bounded below any log poles along the exceptional divisor over $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$ and $\varphi_{\mathcal{V}}$ is uniformly bounded at $q_{j}$. Then we have
$$\liminf _{j \rightarrow \infty}|f|\left(q_{j}\right)>0 .$$
This contradicts the fact that $f\left(q^{\prime}\right)=0$.
\item[(2)] $q^{\prime} \notin \operatorname{LCS}\left(\mathcal{X}_{0}\right)$. Without loss of generality, we can assume that there exists $C>0$ such that for any $x \in B_{d_{\mathbf{Y}}}(q, r) \cap \mathcal{R}_{\mathcal{X}_{0}}$,
$$d_{\chi}\left(x, \operatorname{LCS}\left(\mathcal{X}_{0}\right)\right)>C .$$
Otherwise, there exist a sequence of $x_{j} \in B_{d_{\mathbf{Y}}}(q, r) \cap \mathcal{R}_{\mathcal{X}_{0}}$ such that $x_{j}$ converges to some $x^{\prime} \in \operatorname{LCS}\left(\mathcal{X}_{0}\right)$ in $\left(\mathcal{X}_{0}, \chi_{0}\right)$. This can be reduced to the previous case.
Immediately, we can show the extension $\sigma_{0}^{\prime}$ on $\mathcal{X}_{0}$ from $\sigma_{0}$ does not vanish near $q^{\prime}$ because $\left|\sigma_{0}^{\prime}\right|^{2 / m} e^{\varphi_{\mathcal{V}}}=\Omega_{0} e^{\varphi_{0}}$ and since $\varphi_{\mathcal{V}}\left(q_{j}\right)$ and $\varphi_{0}\left(q_{j}\right)$ are both uniformly bounded for all $j$. Since $\frac{\sigma_{k}\left(q_{j}\right)}{\sigma_{0}\left(q_{j}\right)}=\frac{\sigma_{k}^{\prime}\left(q_{j}\right)}{\sigma_{0}^{\prime}\left(q_{j}\right)}$ converges to $\frac{\sigma_{k}(q)}{\sigma_{0}(q)}, \frac{\sigma_{k}^{\prime}\left(q^{\prime}\right)}{\sigma_{0}^{\prime}\left(q^{\prime}\right)}=\frac{\sigma_{k}(q)}{\sigma_{0}(q)}$. Therefore we also obtain a holomorphic map
$$F^{\prime}=\left(\frac{\sigma_{1}^{\prime}}{\sigma_{0}^{\prime}}, \ldots, \frac{\sigma_{n}^{\prime}}{\sigma_{0}^{\prime}}\right): \mathcal{X}_{0} \backslash\left\{\sigma_{0}^{\prime}=0\right\} \rightarrow \mathbb{C}^{n}, F^{\prime}\left(q^{\prime}\right)=F(q) .$$
Let $U=\left(F^{\prime}\right)^{-1}(\mathcal{V})$, where $\mathcal{V}$ is given by (6.24). Then $U$ is an open set in $\mathcal{X}_{0}$ and $q^{\prime} \in U$.
Since the nonsingular part of $\mathcal{X}_{0}$ is open and dense in Y, there exists a subvariety $D$ of $\mathcal{V}$ such that
$$\left.F^{\prime}\right|_{\left(F^{\prime}\right)^{-1}(\mathcal{V} \backslash D)}:\left(F^{\prime}\right)^{-1}(\mathcal{V} \backslash D) \rightarrow \mathcal{V} \backslash D$$
is biholomorphic.
Recall $\left\{\eta_{0}, \ldots, \eta_{N}\right\}$ gives a projective embedding of $\mathcal{X}_{0}$. There exists a sufficiently small open neighborhood $U_{q^{\prime}}$ of $q^{\prime}$ in $\mathcal{X}_{0}$ such that
$$\gamma=\left(\frac{\eta_{1}}{\sigma_{0}^{\prime}}, \ldots, \frac{\eta_{N}}{\sigma_{0}^{\prime}}\right): U_{q^{\prime}} \rightarrow \mathbb{C}^{N}$$
is a local affine embedding, where we assume $\eta_{0}$ does not vanish on $U_{q^{\prime}}$ (we can always shrink $U_{q^{\prime}}$ ). We can also assume $F^{\prime}\left(U_{q^{\prime}}\right) \subset \mathcal{V}$ by continuity of $F^{\prime}$ since
\end{itemize}
$\sigma_{0}^{\prime}\left(q^{\prime}\right) \neq 0$ and $F^{\prime}\left(q^{\prime}\right) \in \mathcal{V}$. We now consider the map $F^{\prime \prime}$ defined by
$$F^{\prime \prime}=\left(\frac{\sigma_{1}^{\prime}}{\sigma_{0}^{\prime}}, \ldots, \frac{\sigma_{n}^{\prime}}{\sigma_{0}^{\prime}}, \frac{\eta_{1}}{\sigma_{0}^{\prime}}, \ldots, \frac{\eta_{N}}{\sigma_{0}^{\prime}}\right): U_{q^{\prime}} \rightarrow \mathbb{C}^{n} \times \mathbb{C}^{N} .$$
Clearly, $F^{\prime \prime}$ is also an affine embedding of $U_{q^{\prime}}$. We identify $\frac{\eta_{i}}{\sigma_{0}^{\prime}}$ on $\left(F^{\prime}\right)^{-1}(\mathcal{V} \backslash D)$ and $\frac{\eta_{i}}{\sigma_{0}^{\prime}} \circ\left(F^{\prime-1}\right)$ on $\mathcal{V} \backslash D$. Each $\frac{\eta_{i}}{\sigma_{0}^{\prime}}$ is a holomorphic function on $\mathcal{V} \backslash D$ and
$$\sup _{\mathcal{V} \backslash D} \frac{\left(\sum_{i=0}^{N} \eta_{i} \wedge \overline{\eta_{i}}\right)^{1 / m}}{\left|\sigma_{0}^{\prime}\right|^{2 / m}}=\sup _{\mathcal{V} \backslash D} e^{\varphi_{0}-\varphi_{\mathcal{V}}}<\infty$$
Therefore each $\frac{\eta_{i}}{\sigma_{0}^{\prime}}$ is uniformly bounded on $\mathcal{V}$ and it extends to a holomorphic function in $\mathcal{V}$. In particular, each $\frac{\eta_{i}}{\sigma_{0}^{\prime}}$ is a holomorphic function in $\left(\frac{\sigma_{1}^{\prime}}{\sigma_{0}^{\prime}}, \ldots, \frac{\sigma_{n}^{\prime}}{\sigma_{0}^{\prime}}\right)$ on $\mathcal{V}$ for $i=0,1, \ldots, N$. So $F^{\prime \prime}$ is an isomorphism between $U_{q^{\prime}}$ and $F^{\prime}\left(U_{q^{\prime}}\right)$. But $F^{\prime}\left(U_{q^{\prime}}\right)$ lies in the graph of $\mathcal{V}$ of $\left(\frac{\eta_{1}}{\sigma_{0}^{\prime}}, \ldots, \frac{\eta_{N}}{\sigma_{0}^{\prime}}\right)$ in $\mathbb{C}^{n} \times \mathbb{C}^{N}$ and so $F^{\prime}\left(U_{q^{\prime}}\right)$ must be nonsingular. Contradiction. \(\square\)

Corollary 6.1. Let $\mathcal{A}$ be the number of the component of $\mathcal{X}_{0}$ and $\mathcal{B}$ the number of the components of Y. Then
$$\mathcal{A}=\mathcal{B} .$$
Proof. Let $p$ and $p^{\prime}$ be two nonsingular points in distinct components $X$ and $X^{\prime}$ of $\mathcal{X}_{0}$. Then there exists a smooth minimal geodesic $\gamma$ joining $p$ and $p^{\prime}$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ by the geodesic convexity result in [15], and $\gamma$ lies entirely in $\mathcal{R}_{\mathbf{Y}}=\mathcal{R}_{\mathcal{X}_{0}}$. The corollary is proved because $\mathcal{R}_{\mathcal{X}_{0}} \cap\left(X \cup X^{\prime}\right)$ is not connected. \(\square\)

We now prove another version of Schwarz lemma with suitable barriers.
Lemma 6.2. Let $\omega_{0}=\chi_{0}+\sqrt{-1} \partial \bar{\partial} \varphi_{0}$ be the unique Kähler-Einstein current on $\mathcal{X}_{0}$. For any compact set $K \subset \subset \mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right)$, there exists $c=c(K)>0$ such that
$$\omega_{0} \geq c \chi_{0}$$
on $K \cap \mathcal{R}_{\mathcal{X}_{0}}$.
Proof. Since $\mathcal{X}_{0}$ is a semi-log canonical model, by Lemma 3.1, for any $p \in \mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right)$, there exists an effective $\mathbb{Q}$-divisor $G_{p}$ numerically equivalent to $K_{\mathcal{X}_{0}}$ such that $p$ does not lie in $G_{p}$ and the support of $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$ is contained in the support of $G_{p}$. Let $\sigma_{G_{p}}$ be the defining divisor of $G_{p}$ and $h_{\Omega_{0}}$ be the hermitian metric on $G_{p}$ so that $\operatorname{Ric}\left(h_{\Omega_{0}}\right)=\chi_{0}=\left.\chi\right|_{\mathcal{X}_{0}}$.

By similar argument in Lemma 3.1, we can also find an effective $\mathbb{Q}$-divisor $\mathcal{F}$ numerically equivalent to $K_{\mathcal{X}_{0}}$ such that the support of the singular set $\mathcal{S}_{\mathcal{X}_{0}}$ of $\mathcal{X}_{0}$ is contained in the support of $\mathcal{F}$. Let $\sigma_{\mathcal{F}}$ be the defining divisor of $\mathcal{F}$.

Let $\omega_{0}=\chi_{0}+\sqrt{-1} \partial \bar{\partial} \varphi_{0}$ be the unique Kähler-Einstein current on $\mathcal{X}_{0}$. By Lemma 3.6, there exists $C_{1}>0$ and for any $\epsilon>0$, there exists $C_{2}=C_{2}(\epsilon)>0$ such that
$$-C_{2}+\epsilon \log \left|\sigma_{G_{p}}\right|_{h_{\Omega_{0}}}^{2} \leq \varphi_{0} \leq C_{1} .$$

We now define
$$H_{\epsilon, \epsilon^{\prime}}=\log t r_{\omega_{0}}\left(\chi_{0}\right)-3 A \varphi_{0}+\epsilon \log \left|\sigma_{G_{p}}\right|_{h_{\Omega_{0}}}^{2}+\epsilon^{\prime} \log \left|\sigma_{\mathcal{F}}\right|_{h_{\Omega_{0}}}^{2} .$$
By Lemma 4.3, for any sufficiently small $\epsilon^{\prime}>0$, we have
$$\sup _{\mathcal{R}_{\mathcal{X}_{0}}}\left|\sigma_{F}\right|_{h_{\Omega_{0}}}^{2 \epsilon^{\prime}} \operatorname{tr}_{\omega_{0}}\left(\chi_{0}\right)<\infty$$
and $\left|\sigma_{F}\right|_{h_{\Omega_{0}}}^{2 \epsilon^{\prime}} \operatorname{tr}_{\omega_{0}}\left(\chi_{0}\right)$ tends $-\infty$ near the support $\mathcal{F}$.
Straightforward calculations show that on the nonsingular part of $\mathcal{X}_{0}$, for a fixed sufficiently large $A>0$, there exists $C_{3}>0$ such that for all sufficiently small $\epsilon, \epsilon^{\prime}>0$,
$$\Delta_{\omega_{0}} H_{\epsilon, \epsilon^{\prime}} \geq \operatorname{Atr}_{\omega_{0}}\left(\chi_{0}\right)-C_{3} .$$
We can apply the maximum principle for $H_{\epsilon, \epsilon^{\prime}}$ at its maximal point, which must lie in the nonsingular part of $\mathcal{X}_{0}$. By the estimate for $\varphi_{0}$, there exists $C_{4}=C_{4}(\epsilon)>0$ such that
$$H_{\epsilon, \epsilon^{\prime}} \leq C_{4}$$
and so on $\mathcal{R}_{\mathcal{X}_{0}}$, there exists $C_{5}=C_{5}(\epsilon)>0$ such that
$$\operatorname{tr}_{\omega_{0}}\left(\chi_{0}\right) \leq\left|\sigma_{G_{p}}\right|_{h_{\Omega_{0}}}^{-2 \epsilon}\left|\sigma_{\mathcal{F}}\right|_{h_{\Omega_{0}}}^{-2 \epsilon^{\prime}} e^{C_{4}+3 A \varphi_{0}} \leq C_{5}\left|\sigma_{G_{p}}\right|_{h_{\Omega_{0}}}^{-2 \epsilon}\left|\sigma_{\mathcal{F}}\right|_{h_{\Omega_{0}}}^{-2 \epsilon^{\prime}} .$$
Since the constant $C_{5}$ does not depend on $\epsilon^{\prime}$, the lemma is proved by letting $\epsilon^{\prime} \rightarrow 0$. \(\square\)

Lemma 6.3. Let $X$ be a component of $\mathcal{X}_{0}$ and $p \in X \cap \mathcal{R}_{\mathcal{X}_{0}}$. For any $K \subset \subset \mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right)$, there exists $C_{K}>0$ such that for any $q \in X \cap \mathcal{R}_{\mathcal{X}_{0}} \cap K$,
$$d_{\mathbf{Y}}(p, q) \leq C_{K} .$$
Proof. We prove by contradiction. Suppose there exist a sequence of points $q_{j} \in X \cap$ $\mathcal{R}_{\mathcal{X}_{0}} \cap K$ such that
$$d_{\mathbf{Y}}\left(q_{j}, p\right) \rightarrow \infty .$$
We can assume that $q_{j} \rightarrow q$ in $\left(\mathcal{X}_{0}, \chi_{0}\right)$, where $q$ must be a singular point in $\mathcal{X}_{0}$ away from $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$.

Then there exists $r_{0}>0$ such that for all $j, B_{\chi_{0}}\left(q_{j}, r_{0}\right)$, the geodesic ball in $\left(\mathcal{X}_{0}, \chi_{0}\right)$ centered at $q_{j}$ with radius $r_{0}$, lies outside an open neighborhood $U$ of $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$ in $\left(\mathcal{X}_{0}, \chi_{0}\right)$ since $q$ is away from $\operatorname{LCS}\left(\mathcal{X}_{0}\right)$. By Lemma 6.2 and geodesic convexity of $\mathcal{R}_{\mathbf{Y}}=\mathcal{R}_{\mathcal{X}_{0}}$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$, there exists $r_{0}^{\prime}>0$ such that for all $j$,
$$B_{d_{\mathbf{Y}}}\left(q_{j}, 2 r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathbf{Y}} \subset B_{\chi}\left(q_{j}, r_{0}\right)$$
and so
$$B_{d_{\mathbf{Y}}}\left(q_{j}, 2 r_{0}^{\prime}\right) \cap\left(U \cap \mathcal{R}_{\mathcal{X}_{0}}\right)=\phi .$$
Since the total volume of $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ is bounded and $d_{\mathbf{Y}}\left(q_{j}, p\right) \rightarrow \infty$,
$$\epsilon_{j}=\operatorname{Vol}\left(B_{d_{\mathbf{Y}}}\left(q_{j}, 2 r_{0}^{\prime}\right), d_{\mathbf{Y}}\right) \rightarrow 0 .$$

Now we construct the auxiliary smooth function $1 \leq F_{j} \leq A_{j}$ on $\mathcal{X}_{0}$ such that
$$F_{j}=1, \text { on } \mathcal{X}_{0} \backslash\left(B_{d_{\mathbf{Y}}}\left(q_{j}, 2 r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}\right)$$
and
$$F_{j}=A_{j}, \text { on } B_{d_{\mathbf{Y}}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}} .$$

Then we calculate the $L^{1+\delta}$-norm of $F_{j}$ on $\mathcal{X}_{0}$ for some fixed small $\delta>0$. Let $\omega_{0}$ be the unique Kähler-Einstein current on $\mathcal{X}_{0}$ with $\omega_{0}^{n}=e^{\varphi_{0}} \Omega_{0}$. There exists $C_{1}>0$ such that for all $j$,
$$\begin{aligned}
& \int_{\mathcal{X}_{0} \backslash U} F_{j}^{1+\delta} \Omega_{0} \\
\leq & \int_{\mathcal{X}_{0} \backslash\left(U \cup\left(B_{d_{\mathbf{Y}}}\left(q_{j}, 2 r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}\right)\right)} \Omega_{0}+A_{j}^{1+\delta} e^{\sup _{B_{d_{\mathbf{Y}}}\left(q, 2 r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}}\left|\varphi_{0}\right|} \int_{B_{d_{\mathbf{Y}}}\left(q_{j}, 2 r_{0}^{\prime}\right)} d V_{d \mathbf{Y}} \\
\leq & C_{1}+C_{1} A_{j}^{1+\delta} \epsilon_{j}
\end{aligned}$$
We choose
$$A_{j}=\left(\epsilon_{j}\right)^{-\frac{1}{1+\delta}} \rightarrow \infty$$
and so
$$\int_{\mathcal{X}_{0} \backslash U}\left(F_{j}\right)^{1+\delta} \Omega \leq 2 C_{1}$$
for some fixed $\delta>0$ and for all $j$.
Now we consider the family of complex Monge-Ampère equations on $\mathcal{X}_{0}$
$$\left(\chi_{0}+\sqrt{-1} \partial \bar{\partial} \phi_{j}\right)^{n}=e^{\phi_{j}} F_{j} \Omega_{0} .$$
By Lemma 3.8, there exists a solution $\phi_{j}$ solving (6.26). Furthermore, $\phi_{j} \in C^{\infty}\left(\mathcal{R}_{\mathcal{X}_{0}}\right)$ and for any $K \subset \subset \mathcal{X}_{0} \backslash U$, there exists $C_{2}>0$ such that for all $j$,
$$\sup _{B_{d_{\mathbf{Y}}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}}\left|\phi_{j}\right| \leq C_{2} .$$
From equation (6.26), $\chi_{0}+\sqrt{-1} \partial \bar{\partial} \phi_{j}$ is Kähler-Einstein on $B_{d \mathbf{Y}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}$.
We will now prove a local Schwarz lemma. Let $\omega_{j}=\chi_{0}+\sqrt{-1} \partial \bar{\partial} \phi_{j}$ and let $\omega_{0}=$ $\chi_{0}+\sqrt{-1} \partial \bar{\partial} \varphi_{0}$ be unique Kähler-Einstein current on $\mathcal{X}_{0}$. Let
$$r(x)=d_{\mathbf{Y}}(x, p)$$
be the distance function from $x$ to $p$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$. By the geodesic convexity result in [15], for any $x \in \mathcal{R}_{\mathcal{X}_{0}} \cap X$, there exists a smooth geodesic $\gamma$ joining $p$ and $x$, and $\gamma$ lies entirely in $\mathcal{R}_{\mathcal{X}_{0}}$. In particular, $r(x)$ is smooth on $\mathcal{R}_{\mathcal{X}_{0}} \cap X$ except at cut-locus points on smooth minimal geodesics.

We pick an effective divisor $\mathcal{F}$ similarly as in Lemma 6.3 such that $\mathcal{F}$ is numerically equivalent to $K_{\mathcal{X}_{0}}$ and it contains the singular locus $\mathcal{S}_{\mathcal{X}_{0}}$ of $\mathcal{X}_{0}$. We consider the quantity
$$H_{j, \varepsilon}=\log \left(\psi \frac{\omega_{j}^{n}}{\omega_{0}^{n}}\right)-\varepsilon\left(\varphi_{0}-\gamma \log \left|\sigma_{\mathcal{F}}\right|_{h_{\Omega_{0}}}^{2}\right),$$
where $\psi=\phi(r(x))$ is chosen with a smooth cut-off function $\phi$ satisfing
$$\phi(r)=1 \text {, if } r \leq r_{0}^{\prime}, \phi(r)=0 \text {, if } r \geq 2 r_{0}^{\prime}$$
and
$$\phi \geq 0,0 \leq \phi^{-2+\frac{1}{n}}\left(\phi^{\prime}\right)^{2} \leq C, \phi^{-1+\frac{1}{n}}\left|\phi^{\prime \prime}\right| \leq C_{3}$$
for some fixed constant $C_{3}>0$. Since $\varphi_{0}$ is milder than any log pole singularities, $\varphi_{0}-\gamma \log \left|\sigma_{\mathcal{D}}\right|_{h_{\mathcal{D}}}^{2}$ is uniformly bounded below after fixing $\gamma>0$. Immediately, we can conclude that
$$\sup _{\mathcal{X}_{0}} H_{j, \varepsilon}<\infty$$
and $H_{j, \varepsilon}$ tends to $-\infty$ near $G_{p}$. Suppose the maximum of $H_{j, \epsilon}$ is achieved at $q^{\prime}$. Then $q^{\prime} \in\left(\mathcal{R}_{\mathcal{X}_{0}} \backslash \operatorname{Supp}\left(G_{p}\right)\right) \cap \operatorname{Supp}(\psi)$.

If $q^{\prime}$ is not a cut-point of $p$, then $H_{j, \varepsilon}$ is smooth near $q^{\prime}$ and there exists $C_{4}>0$ such that
$$\begin{aligned}
\Delta_{\omega_{0}} H_{j, \varepsilon} & \geq \operatorname{tr}_{\omega_{0}}\left(\omega_{j}\right)+\varepsilon(1-\gamma) \operatorname{tr}_{\omega_{0}}\left(\chi_{0}\right)-n-n \varepsilon-\left(\psi^{-2}|\nabla \psi|_{\omega_{0}}^{2}-\psi^{-1} \Delta_{\omega_{0}} \psi\right) \\
& \geq\left(\frac{\left(\omega_{j}\right)^{n}}{\left(\omega_{0}\right)^{n}}\right)^{1 / n}-C_{4}-C_{4} \psi^{-1 / n} .
\end{aligned}$$
Therefore at the maximum point $q^{\prime}$ of $H_{j, \varepsilon}, \psi\left(\frac{\left(\omega_{j}\right)^{n}}{\left(\omega_{0}\right)^{n}}\right) \leq\left(2 C_{4}\right)^{n}$ and so there exists $C_{5}>0$ such that for all $j$ and $0<\varepsilon<1$,
$$H_{j, \varepsilon} \leq C_{5} .$$

If $q$ is a cut-point of $p$, one can use the following trick of Calabi (c.f. [14]). Let $\gamma$ be a smooth minimizing geodesic joining $p$ and $q$ with $\gamma(0)=p, \gamma(r(q))=q$. Let $p_{\delta}=\gamma(\delta)$ for sufficiently small $\delta>0$. Obviously $q$ is not a cut-point of $p_{\delta}$. Let $r_{\delta}(x)$ be the distance function from $x$ to $p_{\epsilon}$ and $\psi_{\delta}(x)=\phi\left(r_{\delta}(x)+\delta\right)$. Since $r_{\delta}(x)+\delta=r_{\delta}(x)+r_{\delta}(p) \geq r_{p}(x)$, $\psi_{\epsilon}(x) \leq \psi(x)$ and $\psi_{\delta}(q)=\psi(q)$. One now can apply the maximum principle to
$$H_{j, \epsilon, \delta}=\log \left(\psi_{\delta} \frac{\left(\omega_{j}\right)^{n}}{\left(\omega_{0}\right)^{n}}\right)-\epsilon\left(\varphi_{0}-\gamma \log \left|\sigma_{\mathcal{F}}\right|_{h_{\Omega_{0}}}^{2}\right)$$
since the maximum of $H_{\epsilon, \delta}$ is also achieved at $q$. We obtain the same estimate as in the non cut-locus case.

Finally we let $\epsilon \rightarrow 0$. Then there exists $C_{6}>0$ such that for all $j$, we have
$$\sup _{B_{d_{\mathbf{Y}}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}} \frac{\left(\omega_{j}\right)^{n}}{\left(\omega_{0}\right)^{n}} \leq C_{6} .$$
But from the equations for $\varphi_{0}$ and $\varphi_{j}$, we have
$$\sup _{B_{d_{\mathbf{Y}}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}}\left(\frac{\left(\omega_{j}\right)^{n}}{\left(\omega_{0}\right)^{n}}\right)=\sup _{B_{d_{\mathbf{Y}}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}} F_{j} e^{\phi_{j}-\varphi_{0}}=A_{j} \sup _{B_{d_{\mathbf{Y}}}\left(q_{j}, r_{0}^{\prime}\right) \cap \mathcal{R}_{\mathcal{X}_{0}}} e^{\phi_{j}-\varphi_{0}} \rightarrow \infty$$
as $\varphi_{0}$ and $\phi_{j}$ are uniformly bounded on $B_{d_{\mathbf{Y}}}\left(q_{j}, 1\right) \cap \mathcal{R}_{\mathcal{X}_{0}}$. Contradiction. \(\square\)

Lemma 6.4. For any $q_{j} \in \mathcal{R}_{\mathcal{X}_{0}}$ converging to some $q \in \operatorname{LCS}\left(\mathcal{X}_{0}\right)$ in $\left(\mathcal{X}_{0}, \chi_{0}\right)$, we have
$$\lim _{j \rightarrow \infty} d_{\mathbf{Y}}\left(p, q_{j}\right)=\infty .$$
Proof. We prove by contradiction. Suppose there exist a sequence of points $q_{j} \in \mathcal{R}_{\mathcal{X}_{0}} \rightarrow$ $q \in \operatorname{LCS}\left(\mathcal{X}_{0}\right)$ with respect to $\chi_{0}$ and $d_{\mathbf{Y}}\left(q_{j}, p\right) \leq D$ for some fixed $D>0$. We let $q^{\prime}=\lim _{j \rightarrow \infty} q_{j}$ with respect to $d_{\mathbf{Y}}$. Then using the partial $C^{0}$-estimate in Corollary 5.1, there exists a global $L^{2}$-section $\sigma$ on Y such that $\sigma$ is a global section of $m K_{\mathbf{Y}}$ for some $m$ and $\sigma$ does not vanish near $q^{\prime}$. In particular, $|\sigma|_{h_{\mathbf{Y}}}^{2}$ is bounded above and below away from

0 near $q^{\prime}$ in Y , where $h_{\mathbf{Y}}$ is the hermitian metric on $K_{\mathbf{Y}}$ induced from the Kähler-Einstein volume form on $\mathcal{R}_{\mathbf{Y}}$. Hence we can write $h_{\mathbf{Y}}=e^{-\psi}$ for some bounded plurisubharmonic function near $q^{\prime}$ because $\mathbf{Y}$ is an analytic normal space and $-\sqrt{-1} \partial \bar{\partial} \log |\sigma|_{h_{\mathbf{Y}}}^{2}=\sqrt{-1} \partial \bar{\partial} \psi$ is the Kähler-Einstein metric.

On the other hand, $\Omega_{0}=\left(\sum_{i=0}^{N} \eta_{i} \wedge \overline{\eta_{i}}\right)^{1 / m}$ and the Kähler-Einstein volume form is given by $e^{\varphi_{0}} \Omega_{0}$. Each $\eta_{i}$ is a holomorphic section of $m K_{\mathbf{Y}}$ near $q^{\prime}$ on the regular part of Y. Since Y is normal near $q^{\prime}$ by the result of [19] as a local version of [18], $\eta_{i}$ extends to a holomorphic section over $q^{\prime}$ and therefore near $q^{\prime}$. Compare the two volume measures, there exists $C>0$ such that near $q^{\prime}$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$,
$$\varphi_{0}=\psi+\frac{1}{m} \log \left(\frac{|\sigma|^{2}}{\sum_{i=0}^{N}\left(\eta_{i} \wedge \overline{\eta_{i}}\right)}\right) \geq-C .$$
On the other hand, $\varphi_{0}\left(q_{j}\right) \rightarrow-\infty$. Contradiction. \(\square\)

Corollary 6.2. For any $p \in \mathcal{R}_{\mathbf{Y}}$ and any $R \geq 1$, there exists $C=C(p, R)>0$ such that
$$\sup _{B_{d_{\mathbf{Y}}}(p, R) \cap \mathcal{R}_{\mathcal{X}_{0}}}\left|\varphi_{0}\right| \leq C .$$
Proof. We prove by contradiction. Suppose there exist a sequence of points $q_{j} \in B_{d_{\mathbf{Y}}}(p, R) \cap$ $\mathcal{R}_{\mathcal{X}_{0}} \subset \mathbf{Y}$ such that $\varphi_{0}\left(q_{j}\right) \rightarrow-\infty$ as $j \rightarrow \infty$. Then after passing to a subsequence, we can assume that $q_{j}$ converges to $q^{\prime} \in \operatorname{LCS}\left(\mathcal{X}_{0}\right)$ in $\left(\mathcal{X}_{0}, \chi_{0}\right)$ and $q_{j}$ converges to some $q \in B_{d_{\mathbf{Y}}}\left(p_{0}, 2 R\right)$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$. This leads to contradiction because $d_{\mathbf{Y}}\left(p, q_{j}\right) \rightarrow \infty$ from Lemma 6.4. \(\square\)

In conclusion, in each component of $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$, the local boundedness of the Kähler-Einstein potential is equivalent to the boundedness of distance in a uniform way.

\section*{7. Proof of Theorem 1.2}

In this section, we will complete the proof of Theorem 1.2. We first derive a geometric Schwarz lemma.

Lemma 7.1. For any $p \in \mathcal{R}_{\mathcal{X}_{0}}$ and any $R \geq 1$, there exists $C=C(R)>0$ such that
$$\omega_{0} \geq C \chi_{0},$$
on $B_{d_{\mathbf{Y}}}(p, R) \cap R_{\mathbf{Y}}$.
Proof. We let $r(x)$ be the distance function from $x \in \mathbf{Y}$ to $p$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$. We choose the smooth cut-off function $\psi=\phi(r(x))$ satisfying
$$\phi(r)=1, \text { if } r \leq R, \phi(r)=0, \text { if } r \geq 2 R .$$
Furthermore, we can assume that
$$\phi \geq 0,0 \leq \phi^{-1}\left(\phi^{\prime}\right)^{2} \leq C R^{-2},\left|\phi^{\prime \prime}\right| \leq C R^{-2} .$$

We consider the quantity $\operatorname{tr}_{g_{0}}\left(\chi_{0}\right)$ on $\mathcal{R}_{\mathbf{Y}}=\mathcal{R}_{\mathcal{X}_{0}}$. Then straightforward calculations show that there exists $K>0$ such that on $\mathcal{R}_{\mathcal{X}_{0}}$, we have
$$\Delta_{\omega_{0}} \log t r_{\omega_{0}}\left(\chi_{0}\right) \geq-K\left(t r_{\omega_{0}}\left(\chi_{0}\right)\right)-K .$$

Now we choose the same effective divisor $\mathcal{F}$ as in the proof of Lemma 6.3 such that $\mathcal{F}$ is numerically equivalent to $K_{\mathcal{X}_{0}}$ and it contains the singular locus $\mathcal{S}_{\mathcal{X}_{0}}$ of $\mathcal{X}_{0}$. Let $\sigma_{\mathcal{F}}$ be the defining function of $\mathcal{F}$. Then we consider the following quantity
$$H_{\epsilon}=\log \left(\psi t r_{\omega_{0}}\left(\chi_{0}\right)\right)-A \varphi_{0}+\epsilon \log \left|\sigma_{\mathcal{F}}\right|_{h_{\Omega_{0}}}^{2} .$$
There exists $C_{1}>0$ such that
$$\begin{aligned}
& \Delta_{\omega_{0}} H_{\epsilon} \\
= & \Delta_{\omega_{0}}\left(\log t r_{\omega_{0}}\left(\chi_{0}\right)-A \varphi_{0}\right)-\epsilon t r_{\omega_{0}}\left(\chi_{0}\right)-\left(\psi^{-2}|\nabla \psi|^{2}-\psi^{-1} \Delta_{g_{t}} \psi\right) \\
\geq & t r_{\omega_{0}}\left(\chi_{0}\right)-A n-C_{1} \psi^{-1},
\end{aligned}$$
by fixing a sufficiently large $A$. $H_{\epsilon}$ is smooth on $\mathcal{R}_{\mathcal{X}_{0}} \backslash \mathcal{F}$ and by Lemma 4.3, $H_{\epsilon}$ tends to $-\infty$ near $\mathcal{F}$ for any fixed $\epsilon>0$. If $H_{\epsilon}$ achieves its positive maximum at $z_{0}$ in the interior of $B_{d_{\mathbf{Y}}}(p, 2 R)$, then there exists $C_{2}>0$ such that
$$H_{\epsilon}\left(z_{0}\right) \leq C_{2} \sup _{B_{\mathbf{Y}}(p, R) \cap\left(\mathcal{R}_{\mathcal{X}_{0}} \backslash \mathcal{F}\right)}\left|\varphi_{0}\right|$$
Since $\sup _{B_{\mathbf{Y}}(p, R) \cap\left(\mathcal{R}_{\mathcal{X}_{0}} \backslash \mathcal{F}\right)}\left|\varphi_{0}\right|$ is uniformly bounded by Corollary 6.2, there exists $C_{3}>0$ such that
$$\sup _{B_{\mathbf{Y}}(p, R) \cap\left(\mathcal{R}_{\mathcal{X}_{0}} \backslash \mathcal{F}\right)} \operatorname{tr}_{\omega_{0}}\left(\chi_{0}\right) \leq C_{3} .$$
This proves the estimate (7.29). \(\square\)

Corollary 7.1. There exists a unique map
$$\Phi:\left(\mathbf{Y}, d_{\mathbf{Y}}\right) \rightarrow\left(\mathcal{X}_{0}, \chi\right)$$
extending the identity map from $\mathcal{R}_{\mathbf{Y}}$ and $\mathcal{R}_{\mathcal{X}_{0}}$. Furthermore, $\Phi$ is a Lipschitz map and
$$\Phi(\mathbf{Y})=\mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right) .$$
Proof. For any point $p$ in $\mathcal{R}_{\mathbf{Y}}$ and for any $R>0$, by Lemma 7.1, there exists $C_{R}>0$ such that
$$\omega_{0} \geq C_{R} \chi_{0}$$
on $B_{d \mathbf{Y}}(p, R) \cap \mathcal{R}_{\mathcal{X}_{0}}$. The geodesic convexity of $\mathcal{R}_{\mathcal{X}_{0}}$ in Y implies that the identity map uniquely extends from $\mathcal{R}_{\mathbf{Y}}$ to $\mathbf{Y}$ and $\Phi$ is locally Lipschitz. Therefore any Cauchy sequence of points in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ must also be a Cauchy sequence in $\left(\mathcal{X}_{0}, \chi_{0}\right)$. The last statement of the corollary follows from the distance estimate in Lemma 6.3 and Lemma 6.4. \(\square\)

Lemma 7.2. $\Phi$ is injective.
Proof. For any two distinct points $q_{1}, q_{2} \in \mathbf{Y}$, the partial $C^{0}$-estimates in Corollary 5.1 allow us to construct two holomophic sections of $m K_{\mathbf{Y}}$ on $\mathbf{Y}$ for some sufficiently large $m$ such that
$$\left|\sigma_{1}\left(q_{1}\right)\right|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2} \geq 2\left|\sigma_{2}\right|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2}\left(q_{1}\right) \neq 0,\left|\sigma_{2}\left(q_{2}\right)\right|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2} \geq 2\left|\sigma_{1}\right|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2}\left(q_{2}\right) \neq 0$$
and
$$\int_{\mathbf{Y}}\left|\sigma_{i}\right|_{\left(h_{\mathbf{Y}}\right)^{m}}^{2} d V_{d_{\mathbf{Y}}}<\infty, i=1,2 .$$

Both $\sigma_{1}$ and $\sigma_{2}$ extend uniquely to pluricanonical sections $\sigma_{1}^{\prime}$ and $\sigma_{2}^{\prime}$ on $\mathcal{X}_{0}$. Then we consider the meromorphic function
$$f=\frac{\sigma_{2}}{\sigma_{1}} \text { on } \mathbf{Y}, f^{\prime}=\frac{\sigma_{2}^{\prime}}{\sigma_{1}^{\prime}} \text { on } \mathcal{X}_{0} .$$
Then $f=\Phi^{*}\left(f^{\prime}\right)=f^{\prime}(\Phi) . \quad f$ and $f^{\prime}$ are holomorphic near $q_{1}, q_{2}$ and $\Phi\left(q_{1}\right), \Phi\left(q_{2}\right)$ respectively and
$$f\left(q_{1}\right) \leq 1 / 2, f\left(q_{2}\right) \geq 2,$$
Therefore $f^{\prime}\left(\Phi\left(q_{1}\right)\right) \neq f^{\prime}\left(\Phi\left(q_{2}\right)\right)$ and so
$$\Phi\left(q_{1}\right) \neq \Phi\left(q_{2}\right) .$$ \(\square\)

Corollary 7.2. Y is homeomorphic to $\mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right)$.
Proof. It suffices to show that $\Phi^{-1}: \mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right) \rightarrow \mathbf{Y}$ is continuous. Let $\left\{x_{j}\right\}_{j=1}^{\infty}$ a sequence of points in a fixed component of $\mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right)$ such that $x_{j} \rightarrow x_{\infty} \in \mathcal{X}_{0} \backslash \operatorname{LCS}\left(\mathcal{X}_{0}\right)$ with respect to $\chi_{0}$. By Lemma 6.3, there exists a base point $p \in \mathcal{R}_{\mathbf{Y}}$ and $C>0$ such that for all $j$,
$$d_{\mathbf{Y}}\left(p, x_{j}\right) \leq C .$$
Let $y_{j_{l}}$ be any convergent subsequence of $y_{j}=\Phi^{-1}\left(x_{j}\right)$ in $\left(\mathbf{Y}, d_{\mathbf{Y}}\right)$ and let $y_{\infty}$ be the limit point. Then by continuity of $\Phi$, we have
$$x_{\infty}=\lim _{l_{j} \rightarrow \infty} x_{l j}=\lim _{l_{j} \rightarrow \infty} \Phi\left(y_{l_{j}}\right)=\Phi\left(y_{\infty}\right) .$$
Therefore $y_{\infty}$ is uniquely determined and $y_{\infty}=\Phi^{-1}\left(x_{\infty}\right)$. \(\square\)

We have now completed the proof of Theorem 1.2 by combining all the previous results we obtained. Theorem 1.3 immediately follows from Theorem 1.2 by using the algebraic result of Hacon and Xu [23].
