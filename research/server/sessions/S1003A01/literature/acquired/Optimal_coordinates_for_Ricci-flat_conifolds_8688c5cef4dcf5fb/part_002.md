It remains to compute the eigenvalues and eigensections generated by smooth functions on $\widehat{M}$. Let $w \in C^{\infty}(\widehat{M})$ be such that $\widehat{\Delta} w=\lambda w$. Then,
$$r^{\xi_{ \pm}(\lambda)} w \in \operatorname{ker}(\bar{\Delta}),$$
and hence
$$r^{\xi_{ \pm}(\lambda)} w \cdot \bar{g}=r^{\xi_{ \pm}(\lambda)} w\left(d r \otimes d r+r^{2} \widehat{g}\right) \in \operatorname{ker}\left(\bar{\Delta}_{L}\right)$$
Due to the structure of the Lichnerowicz Laplacian, we get for
$$h_{1}(w):=\left.r^{\xi_{ \pm}(\lambda)} w \cdot \bar{g}\right|_{\{1\} \times \widehat{M}}=w(d r \otimes d r+\widehat{g})$$
that
$$\square_{L} h_{1}(w)=\lambda \cdot h_{1}(w) .$$
Finally, it remains to consider the $\bar{g}$-traceless eigensections generated by functions. For $v \in C^{\infty}(\widehat{M})$ with $\widehat{\Delta}_{0} v=\lambda v$ and $\alpha, \beta \in \mathbb{R}$, we first make the ansatz
$$\bar{\omega}=r^{\beta}(\alpha v d r+r d v) .$$
Using Lemma 3.14, we get
$$\begin{aligned}
& \bar{\delta}^{*}\left(r^{\beta} v d r\right)=r^{\beta-1}\left(v r^{2} \widehat{g}+\beta v d r \otimes d r+\frac{1}{2} r \widehat{\nabla} v \odot d r\right) \\
& \bar{\delta}^{*}\left(r^{\beta} r d v\right)=r^{\beta-1}\left(r^{2} \widehat{\nabla}^{2} v+\frac{1}{2}(\beta-1) r \widehat{\nabla} v \odot d r\right)
\end{aligned}$$
and we conclude the trace-free part of the symmetrized covariant derivative equals
$$\begin{aligned}
\stackrel{\circ}{\delta}^{*} \bar{\omega}= & \bar{\delta}^{*} \bar{\omega}+\frac{1}{n} \bar{\delta} \bar{\omega} \\
= & r^{\beta-1}\left[r^{2} \widehat{\nabla}^{2} v+\frac{\lambda}{n} v\left(d r \otimes d r+r^{2} \widehat{g}\right)+\frac{\alpha(\beta-1)}{n} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.+\frac{1}{2}(\alpha+\beta-1) r \widehat{\nabla} v \odot d r\right] \\
= & r^{\beta-1}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda+(n-1) \alpha(\beta-1)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.+\frac{1}{2}(\alpha+\beta-1) r \widehat{\nabla} v \odot d r\right]
\end{aligned}$$
We now want to find $\alpha, \beta \in \mathbb{R}$ such that $\bar{\omega} \in \operatorname{ker}\left(\bar{\Delta}_{1}\right)$. By Proposition 3.11, we have
$$\square_{1}\left(\xi_{ \pm}(\lambda) v d r+d v\right)=\lambda_{i, \pm}^{(1)}\left(\xi_{ \pm}(\lambda) v d r+d v\right), \quad \lambda_{ \pm}^{(1)}:=\eta\left(\xi_{ \pm}(\lambda)-1\right) .$$
Since the connection Laplacian $\widehat{\Delta}_{1}$ on one-forms is conical, the forms
$$\begin{array}{ll}
r^{\xi_{+}^{(1)}\left(\lambda_{+}\right)}\left(\xi_{+}(\lambda) v d r+r d v\right), & r^{\xi_{-}^{(1)}\left(\lambda_{+}\right)}\left(\xi_{+}(\lambda) v d r+r d v\right), \\
r^{\xi_{+}^{(1)}\left(\lambda_{-}\right)}\left(\xi_{-}(\lambda) v d r+r d v\right), & r^{\xi_{-}^{(1)}\left(\lambda_{-}\right)}\left(\xi_{-}(\lambda) v d r+r d v\right),
\end{array}$$
are all harmonic on $\bar{M}$. Due to the previous computations, the trace-free parts of their symmetrized covariant derivatives are given by the tensors
$$\begin{aligned}
\bar{h}_{2}(v):= & r^{\xi_{+}\left(\lambda_{+}\right)-1}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda+(n-1) \xi_{+}(\lambda)\left(\xi_{+}\left(\lambda_{+}^{(1)}\right)-1\right)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.+\frac{1}{2}\left(\xi_{+}(\lambda)+\xi_{+}\left(\lambda_{+}^{(1)}\right)-1\right) r \widehat{\nabla} v \odot d r\right], \\
\bar{h}_{3}(v):= & r^{\xi_{-}\left(\lambda_{-}^{(1)}\right)-1}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda+(n-1) \xi_{-}(\lambda)\left(\xi_{-}\left(\lambda_{-}^{(1)}\right)-1\right)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.+\frac{1}{2}\left(\xi_{-}(\lambda)+\xi_{-}\left(\lambda_{-}^{(1)}\right)-1\right) r \widehat{\nabla} v \odot d r\right], \\
\bar{h}_{4}(v):= & r^{\xi_{+}\left(\lambda_{-}^{(1)}\right)-1}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda+(n-1) \xi_{-}(\lambda)\left(\xi_{+}\left(\lambda_{-}^{(1)}\right)-1\right)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.+\frac{1}{2}\left(\xi_{-}(\lambda)+\xi_{+}\left(\lambda_{-}^{(1)}\right)-1\right) r \widehat{\nabla} v \odot d r\right], \\
\bar{h}_{5}(v):= & r^{\xi_{-}\left(\lambda_{+}^{(1)}\right)-1}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda+(n-1) \xi_{+}(\lambda)\left(\xi_{-}\left(\lambda_{+}^{(1)}\right)-1\right)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.+\frac{1}{2}\left(\xi_{+}(\lambda)+\xi_{-}\left(\lambda_{+}^{(1)}\right)-1\right) r \widehat{\nabla} v \odot d r\right],
\end{aligned}$$
which are, by the commutation rules (2.11), all in the kernel of the Lichnerowicz Laplacian. Simple manipulations yield
$$\begin{array}{ll}
\xi_{+}\left(\lambda_{+}^{(1)}\right)=\xi_{+}(\lambda)-1, & \xi_{-}\left(\lambda_{+}^{(1)}\right)=\xi_{-}(\lambda)+1, \\
\xi_{-}\left(\lambda_{-}^{(1)}\right)=\xi_{-}(\lambda)-1, & \xi_{+}\left(\lambda_{-}^{(1)}\right)=\xi_{+}(\lambda)+1,
\end{array}$$
and therefore,
$$\begin{array}{ll}
\xi_{+}\left(\lambda_{+}^{(1)}\right)-1=\xi_{+}(\lambda)-2, & \xi_{-}\left(\lambda_{-}^{(1)}\right)-1=\xi_{-}(\lambda)-2, \\
\xi_{+}\left(\lambda_{-}^{(1)}\right)-1=\xi_{+}(\lambda), & \xi_{-}\left(\lambda_{+}^{(1)}\right)-1=\xi_{-}(\lambda) .
\end{array}$$
This implies that
$$\begin{aligned}
\xi_{-}(\lambda)\left(\xi_{+}\left(\lambda_{-}^{(1)}\right)-1\right) & =\xi_{+}(\lambda)\left(\xi_{-}\left(\lambda_{+}^{(1)}\right)-1\right)=\xi_{+}(\lambda) \cdot \xi_{-}(\lambda)=-\lambda, \\
\xi_{-}(\lambda)+\xi_{+}\left(\lambda_{-}^{(1)}\right)-1 & =\xi_{+}(\lambda)+\xi_{-}\left(\lambda_{+}^{(1)}\right)-1=\xi_{+}(\lambda)+\xi_{-}(\lambda)=2-n .
\end{aligned}$$
Furthermore, we compute
$$\begin{aligned}
\lambda+(n-1) \xi_{ \pm}(\lambda)\left(\xi_{ \pm}(\lambda)-2\right) & =\lambda+(n-1) \xi_{ \pm}(\lambda)\left(\xi_{ \pm}(\lambda)+n-2-n\right) \\
& =\lambda+(n-1) \lambda-(n-1) n \xi_{ \pm}(\lambda)=n\left(\lambda-(n-1) \xi_{ \pm}(\lambda)\right),
\end{aligned}$$
therefore
$$\begin{aligned}
\bar{h}_{2}(v)= & r^{\xi_{+}(\lambda)-2}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda-(n-1) \xi_{+}(\lambda)}{n-1} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.\quad+\left(\xi_{+}(\lambda)-1\right) r \widehat{\nabla} v \odot d r\right] \\
\bar{h}_{3}(v)= & r^{\xi_{-}(\lambda)-2}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\frac{\lambda-(n-1) \xi_{-}(\lambda)}{n-1} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)\right. \\
& \left.\quad+\left(\xi_{-}(\lambda)-1\right) r \widehat{\nabla} v \odot d r\right] \\
\bar{h}_{4}(v)= & r^{\xi_{+}(\lambda)}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)-\frac{\lambda(n-2)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)-\frac{1}{2}(n-2) r \widehat{\nabla} v \odot d r\right], \\
\bar{h}_{5}(v)= & r^{\xi_{-}(\lambda)}\left[r^{2}\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)-\frac{\lambda(n-2)}{n(n-1)} v\left((n-1) d r \otimes d r-r^{2} \widehat{g}\right)-\frac{1}{2}(n-2) r \widehat{\nabla} v \odot d r\right] .
\end{aligned}$$

Because the Lichnerowicz Laplacian is conical, we get that the three tensors
$$\begin{aligned}
\widehat{h}_{2}(v): & =\left.\bar{h}_{2}(v)\right|_{\{1\} \times \widehat{M}} \\
& =\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\left(\frac{\lambda}{n-1}-\xi_{+}(\lambda)\right) v((n-1) d r \otimes d r-\widehat{g})+\left(\xi_{+}(\lambda)-1\right) \widehat{\nabla} v \odot d r, \\
\widehat{h}_{3}(v): & =\left.\bar{h}_{3}(v)\right|_{\{1\} \times \widehat{M}} \\
& =\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)+\left(\frac{\lambda}{n-1}-\xi_{-}(\lambda)\right) v((n-1) d r \otimes d r-\widehat{g})+\left(\xi_{-}(\lambda)-1\right) \widehat{\nabla} v \odot d r, \\
\widehat{h}_{4}(v): & =\left.\bar{h}_{4}(v)\right|_{\{1\} \times \widehat{M}} \\
& =\left(\widehat{\nabla}^{2} v+\frac{\lambda}{n-1} v \widehat{g}\right)-\frac{\lambda(n-2)}{n(n-1)} v((n-1) d r \otimes d r-\widehat{g})-\frac{1}{2}(n-2) \widehat{\nabla} v \odot d r \\
& =\left.\bar{h}_{5}(v)\right|_{\{1\} \times \widehat{M}}
\end{aligned}$$
are eigensections of $\square_{L}$ with eigenvalues
$$\square_{L} \widehat{h}_{2}(v)=\eta\left(\xi_{+}(\lambda)-2\right) \widehat{h}_{2}(v), \quad \square_{L} \widehat{h}_{3}(v)=\eta\left(\xi_{-}(\lambda)-2\right) \widehat{h}_{2}(v), \quad \square_{L} \widehat{h}_{4}(v)=\lambda \widehat{h}_{4}(v) .$$
This finishes the discussion of eigenvalues $\lambda>0$. Recall that if $\lambda=0$, one has $\xi_{+}(\lambda)=0$ and $\xi_{-}(\lambda)=2-n$; moreover, $v$ is constant and all its covariant derivatives vanish. In this case, the only nonvanishing tensor of the above is
$$\widehat{h}_{3}(v)=(n-2) v((n-1) d r \otimes d r-\widehat{g})$$
and the corresponding eigenvalue is $\eta\left(\xi_{-}(\lambda)-2\right)=\eta(-n)=(-n)(-n+n-2)=2 n$. By now we have constructed all eigenvalues and eigensections that appear in the assertion of the theorem. Using the natural identification
$$C^{\infty}\left(\left.S^{2} \bar{M}\right|_{\{1\} \times \widehat{M}}\right)=C^{\infty}(\widehat{M}) \oplus C^{\infty}\left(T^{*} \widehat{M}\right) \oplus C^{\infty}\left(S^{2} \widehat{M}\right)$$
together with the decompositions (3.8) and (3.14), we get that the $L^{2}$-span of the eigensections we have constructed equals all of $L^{2}\left(\left.S^{2} \bar{M}\right|_{\{1\} \times \widehat{M}}\right)$. \(\square\)

Corollary 3.18. Let $\bar{h} \in \operatorname{ker}\left(\bar{\Delta}_{\bar{L}}\right)$, not necessarily defined on all of $\bar{M}$. Let $\nu_{i}, i \in \mathbb{N}$ be the eigenvalues of $\square_{L}$ and define

\subsection*{3.5 A simple proof of linear stability}

Recall that the Hardy inequality states that
$$\inf _{\varphi \in C_{\mathrm{cs}}^{\infty}((0, \infty))} \frac{\int_{0}^{\infty}\left(\partial_{r} \varphi\right)^{2} r^{n-1} d s}{\int_{0}^{\infty} \varphi^{2} r^{n-3} d s}=\frac{(n-2)^{2}}{4} .$$

Lemma 3.19. Let $\bar{\Delta}_{\bar{V}}$ be a conical self-adjoint Laplace type operator acting on sections of a vector bundle $\bar{V}$ over $\bar{M}$. Let $\square_{\widehat{V}}$ be its tangential operator. Then,
$$\bar{\Delta}_{\bar{V}} \geq 0$$
if and only if
$$\square_{\widehat{V}} \geq-\frac{(n-2)^{2}}{4} .$$
Proof. Let $\bar{u} \in C^{\infty}(\bar{V})$ be of the form for $\bar{u}=\varphi(r) u \in C^{\infty}(\bar{V})$ with some $\varphi \in C^{\infty}((0, \infty))$ and an eigensection $u \in E\left(\square_{\widehat{V}}, \nu\right)$ normalized such that $\|u\|_{L^{2}}=1$. By using (3.5) and integrating by parts in $r$, we can write
$$\begin{aligned}
\left(\bar{\Delta}_{\bar{V}} \bar{u}_{i}, \bar{u}_{i}\right)_{L^{2}(\bar{g})} & =\int_{0}^{\infty} \int_{\widehat{M}}\left(\left|\partial_{r} \varphi\right|^{2}|u|^{2}+r^{-2} \varphi^{2}\left\langle\square_{\widehat{V}} u, u\right\rangle\right) r^{n-1} d V_{\widehat{g}} d r \\
& =\int_{0}^{\infty}\left(\left|\partial_{r} \varphi\right|^{2}+\nu r^{-2} \varphi^{2}\right) r^{n-1} d r
\end{aligned}$$
Because the space of finite linear combinations of such sections is $H^{1}$-dense in $C^{\infty}(\bar{E})$, the result now follows from the Hardy inequality. \(\square\)

Using this result, we can now give a simple proof of Theorem 1.2
Proof of Theorem 1.2. Recall from the beginning of Subsection 3.3, that $\mu_{i} \geq(n-2)$ for $i \in \mathbb{N}$. Moreover, by the Lichnerowicz-Obata eigenvalue inequality [Oba62], we get $\lambda_{i} \geq(n-1)$ for $i \in \mathbb{N}$. By definition of the numbers $\mu_{i, \pm}^{(1)}, \lambda_{i, \pm}^{(2)}$ in Theorem 3.15, we obtain
$$\begin{aligned}
& \mu_{i, \pm}^{(1)} \geq \eta\left(\xi_{+}(n-1)-1\right)=0>-\frac{(n-2)^{2}}{4}, \\
& \lambda_{i, \pm}^{(2)} \geq \eta\left(\xi_{+}(n-1)-2\right)=3-n>-\frac{(n-2)^{2}}{4} .
\end{aligned}$$
Therefore, the assertion directly follows from Lemma 3.19 and Theorem 3.15 \(\square\)

\subsection*{3.6 Decay of perturbations satisfying the Bianchi gauge}

In the previous subsection, we have established possible growth and decay rates for homogeneous solutions of the equation $\bar{\Delta}_{L} \bar{h}=0$. However, not all of them are relevant for our further considerations. Instead, we will later only need to consider the solutions which additionally satisfy the equation $B_{\bar{g}}(\bar{h})=\left(\bar{\delta}+\frac{1}{2} d \circ \overline{\operatorname{tr}}\right)(\bar{h})=0$ because by (2.8) and (2.9), this implies
$$\left.\frac{d}{d t}\right|_{t=0} \operatorname{Ric}_{\bar{g}+t \bar{h}}=0,\left.\quad \frac{d}{d t}\right|_{t=0} V(\bar{g}+t \bar{h}, \bar{g})=0,$$
so Ricci-flatness and the Bianchi gauge are both preserved at a linear level. For the second main theorem, it turns out that we can also exclude the solutions which are given by Lie derivatives. Recall that for each eigenvalue $\nu \in \operatorname{spec}\left(\square_{L}\right)$ with $\nu \neq-\frac{(n-2)^{2}}{4}$ and each eigensection $h \in E\left(\square_{L}, \nu\right)$, we have two tensors $\bar{h}_{ \pm}:=r^{\xi_{ \pm}(\nu)} h \in \operatorname{ker}\left(\bar{\Delta}_{L}\right)$. If $\nu=-\frac{(n-2)^{2}}{4}$, we use the notation $\bar{h}_{+}=r^{-\frac{n-2}{2}} \log (r) h$ and $\bar{h}_{-}=r^{-\frac{n-2}{2}} h$. Throughout this subsection, we also keep the notation from Theorem 3.15

Proposition 3.20. Let $\nu$ be an eigenvalue of $\square_{L}, h \in E\left(\square_{L}, \nu\right)$ and $\bar{h}_{ \pm}$as above. Then, the equation
$$B_{\bar{g}}\left(\bar{h}_{ \pm}\right)=0$$
is satisfied by the following tensors $\bar{h}_{ \pm}$:

\item[(iv)] If $\nu=\lambda_{i,+}^{(2)}$ for some $i \in \mathbb{N}$, then $\bar{h}_{+}$solves (3.23) but $\bar{h}_{-}$does not.

\item[(viii)] If $\nu=\lambda_{0,-}^{(2)}=2 n$, then $\bar{h}_{-}$solves (3.23) but $\bar{h}_{+}$does not.

Furthermore, only in the cases (i) and (iv), the tensors $\bar{h}_{ \pm}$which solve (3.23) are not given by Lie derivatives of the metric $\bar{g}$ along any vector fields.

$$B_{\bar{g}}\left(\bar{h}_{-}\right)=\bar{\delta}\left(\bar{h}_{-}\right) \neq 0 .$$
\end{itemize}

\item[(vi)] If $h \in E\left(\square_{L}, \lambda\right)$, then
$$h=\stackrel{\circ}{\nabla}^{2} v+\frac{\lambda(n-2)}{n(n-1)} v \stackrel{\circ}{\widehat{g}}-\frac{1}{2}(n-2) \widehat{\nabla} v \odot d r+w \cdot \bar{g}$$
with $v, w \in E(\widehat{\Delta}, \lambda)$. Let us set $w=0$ for the moment. Then we have
$$\begin{aligned}
\bar{h}_{+} & =r^{\xi_{+}(\lambda)}\left(\stackrel{\circ}{\widehat{\nabla}}^{2} v+\frac{\lambda(n-2)}{n(n-1)} v \stackrel{\circ}{\widehat{g}}-\frac{1}{2}(n-2) \widehat{\nabla} v \odot d r\right) \\
& =\bar{\delta}^{*}\left(r^{\xi_{+}\left(\lambda_{i}\right)+1}\left(\xi_{-}(\lambda) v d r+r d v\right)\right)
\end{aligned}$$
Observe that $\bar{h}_{+}$is trace free. Let us now abbreviate $\bar{\omega}=r^{\xi_{+}(\lambda)+1}\left(\xi_{-}(\lambda) v d r+r d v\right)$ and recall that $\bar{\Delta}_{1}(\bar{\omega})=0$. Then we get
$$B_{\bar{g}} \bar{\omega}=\overline{\delta \delta}^{*} \bar{\omega}=\frac{1}{2} \bar{\Delta}_{1} \bar{\omega}+\left(\frac{1}{2}-\frac{1}{n}\right) \overline{\nabla \delta} \bar{\omega}=\frac{n-2}{2 n} d \bar{\delta} \bar{\omega} .$$
We furthermore compute
$$\begin{aligned}
\bar{\delta}(\bar{\omega}) & =\bar{\delta}\left(r^{\xi_{+}(\lambda)-\xi_{-}(\lambda)+2} r^{\xi_{-}(\lambda)-1}\left(\xi_{-}(\lambda) v d r+r d v\right)\right) \\
& =\bar{\delta}\left(r^{\xi_{+}(\lambda)-\xi_{-}(\lambda)+2} d\left(r^{\xi_{-}(\lambda)} v\right)\right) \\
& =-\left\langle d\left(r^{\xi_{+}(\lambda)-\xi_{-}(\lambda)+2}\right), d\left(r^{\xi_{-}(\lambda)} v\right)\right\rangle_{\bar{g}}+r^{\xi_{+}(\lambda)-\xi_{-}(\lambda)+2} \bar{\Delta}\left(r^{\xi_{-}(\lambda)} v\right) \\
& =\left(\xi_{-}(\lambda)-\xi_{+}(\lambda)+2\right) \xi_{-}(\lambda) r^{\xi_{+}(\lambda)} v
\end{aligned}$$
and consequently
$$B_{\bar{g}}\left(\bar{h}_{+}\right)=\frac{n-2}{2 n} d \bar{\delta}(\bar{\omega})=\frac{n-2}{2 n}\left(\xi_{-}(\lambda)-\xi_{+}(\lambda)+2\right) \xi_{-}(\lambda) d\left(r^{\xi_{+}(\lambda)} v\right) .$$
On the other hand, if $v=0, \bar{h}_{+}=\left(r^{\xi_{+}(\lambda)} w\right) \cdot \bar{g}$ and
$$B_{\bar{g}}\left(\bar{h}_{+}\right)=\left(-1+\frac{n}{2}\right) d\left(r^{\xi_{+}(\lambda)} w\right)=\frac{n-2}{2} d\left(r^{\xi_{+}(\lambda)} w\right) .$$
Now we assume that both $v$ and $w$ are nonzero. From summing up the two subcases we considered before, we see that
$$B_{\bar{g}}\left(\bar{h}_{+}\right)=0$$
if and only if
$$n w=\left(\xi_{+}(\lambda)-\xi_{-}(\lambda)-2\right) \xi_{-}(\lambda) v$$
The discussion for $\bar{h}_{-}$is completely analogous, with interchanging the positions of + and - everywhere in the calculation.
\item[(vii)] If $h \in E\left(\square_{L}, \lambda_{0}=0\right)$, then $\bar{h}_{+}=\alpha \cdot \bar{g}$ and $\bar{h}_{-}=\alpha r^{2-n} \bar{g}$, where $\alpha \in \mathbb{R}$. In the first case, $\bar{h}_{+}$is parallel and hence solves (3.23) whereas in the second case
$$B_{\bar{g}}\left(\bar{h}_{-}\right)=\alpha \frac{n-2}{2} d\left(r^{2-n}\right)=\frac{\alpha}{2}(n-2)^{2} r^{1-n} d r \neq 0 .$$
\end{itemize}
\begin{itemize}
\item[(viii)] If $h \in E\left(\square_{L}, \lambda_{0_{-}}=2 n\right)$, then $h=\alpha \cdot(n-2) \cdot((n-1) d r \otimes d r-\widehat{g})$ for some $\alpha \in \mathbb{R}$ and
$$\bar{h}_{-}=\alpha \stackrel{\circ}{\delta}^{*}\left((2-n) r^{1-n} d r\right)=\alpha \stackrel{\circ}{\nabla}^{2} r^{2-n}=\alpha \bar{\nabla}^{2} r^{2-n}=\alpha \bar{\delta}^{*}\left(d\left(r^{2-n}\right)\right) .$$
Because $r^{2-n}$ is a harmonic function, we easily conclude
$$\bar{\delta}\left(\stackrel{\circ}{\nabla}^{2} r^{2-n}\right)=\bar{\delta}\left(\bar{\nabla}^{2} r^{2-n}\right)=\bar{\Delta}_{1} d\left(r^{2-n}\right)=0$$
and so, $\bar{h}_{-}$solves (3.23) because it is a TT-tensor. On the other hand, $\bar{h}_{+}$is trace free but not divergence free, since
$$\bar{h}_{+}=r^{n+2} \bar{h}_{-},$$
which implies
$$\bar{\delta}\left(\bar{h}_{+}\right)=\bar{\delta}\left(r^{n+2} \bar{h}_{-}\right)=(n+2) r^{n+1} \bar{h}_{-}(d r)+r^{n+2} \overline{\delta h}_{-}=\alpha(n+2)(n-2)(n-1) r d r \neq 0 .$$
\end{itemize}
To finish the proof, we recall that the cases (ii)-(v) and (viii), the tensors $\bar{h}_{ \pm}$which are in the kernel of $B_{\bar{g}}$ are of the form $\bar{h}_{ \pm}=\bar{\delta}^{*}\left(\bar{\omega}_{ \pm}\right)=\frac{1}{2} \mathcal{L}_{\left(\bar{\omega}_{ \pm}\right)^{\#}} \bar{g}$. This is also the case for the metric (case (vii)), since $\bar{g}=\bar{\delta}^{*}(r d r)=\frac{1}{2} \mathcal{L}_{r \partial_{r}} \bar{g}$. \(\square\)

Let us label the indicial set of $\bar{\Delta}_{L}$ by
$$\begin{aligned}
E_{L}:=\{ & \left.\xi_{ \pm}\left(\kappa_{i}\right), \xi_{ \pm}\left(\mu_{i}+1\right)-1, \xi_{ \pm}\left(\mu_{i}+1\right)+1, \xi_{ \pm}\left(\lambda_{i}\right)-2, \xi_{ \pm}\left(\lambda_{i}\right), \xi_{ \pm}\left(\lambda_{i}\right)+2 \mid i \in \mathbb{N}\right\} \\
& \cup\{-n, 2-n, 0,2\}
\end{aligned}$$
and recall from Corollary 3.18 the notation
$$\xi_{+}^{L}:=\min \operatorname{Re}\left(E_{L}\right) \cap(0, \infty), \quad \xi_{-}^{L}:=\min \operatorname{Re}\left(-E_{L}\right) \cap(0, \infty) .$$
We denote the indicial set of $\bar{\Delta}_{L}$ on tensors satisfying the linearized Bianchi gauge by
$$E_{B}=\left\{\xi_{ \pm}\left(\kappa_{i}\right), \xi_{ \pm}\left(\mu_{i}+1\right)-1, \xi_{ \pm}\left(\lambda_{i}\right)-2, \xi_{ \pm}\left(\lambda_{i}\right) \mid i \in \mathbb{N}\right\} \cup\{-n, 0\}$$
and set
$$\xi_{+}^{B}:=\min \operatorname{Re}\left(E_{B}\right) \cap(0, \infty), \quad \xi_{-}^{B}:=\min \operatorname{Re}\left(-E_{B}\right) \cap(0, \infty) .$$
As in the introduction, we label the subset of $E_{B}$ not corresponding to Lie derivatives by
$$E:=\left\{\xi_{ \pm}\left(\kappa_{i}\right), \xi_{ \pm}\left(\lambda_{i}\right) \mid i \in \mathbb{N}\right\} \cup\{0\} .$$
and we set
$$\xi_{+}:=\min \operatorname{Re}(E) \cap(0, \infty), \quad \xi_{-}:=\min \operatorname{Re}(-E) \cap(0, \infty) .$$
Obviously, we have
$$\xi_{+} \geq \xi_{+}^{B} \geq \xi_{+}^{L}, \quad \xi_{-} \geq \xi_{-}^{B} \geq \xi_{-}^{L} .$$

Corollary 3.21. Let $\bar{h} \in \operatorname{ker}\left(\bar{\Delta}_{L}\right) \cap \operatorname{ker}\left(B_{\bar{g}}\right)$, not necessarily defined on all of $\bar{M}$. Then we have the following:
\begin{itemize}
\item[(i)] If $\bar{h}$ is defined on $(0, \epsilon) \times \widehat{M}$ for some $\epsilon>0$ and $|\bar{h}| \rightarrow 0$ as $r \rightarrow 0$, then $\bar{h}=\mathcal{O}_{\infty}\left(r^{\xi_{+}^{B}}\right)$ as $r \rightarrow 0$.
\item[(ii)] If $\bar{h}$ is defined on $(R, \infty) \times \widehat{M}$ for some $R>0$ and $|\bar{h}| \rightarrow 0$ as $r \rightarrow \infty$, then we divide into two subcases:
\begin{itemize}
\item[] (iia) If $\operatorname{spec}\left(\square_{L}\right) \cap\left[-\frac{(n-2)^{2}}{4}, 0\right) \neq\left\{-\frac{(n-2)^{2}}{4}\right\}$, then $\bar{h}=\mathcal{O}_{\infty}\left(r^{-\xi_{-}^{B}}\right)$ as $r \rightarrow \infty$.
\item[] (iib) If $\operatorname{spec}\left(\square_{L}\right) \cap\left[-\frac{(n-2)^{2}}{4}, 0\right)=\left\{-\frac{(n-2)^{2}}{4}\right\}$, then $\bar{h}=\mathcal{O}_{\infty}\left(r^{-\frac{n-2}{2}} \log (r)\right)$ as $r \rightarrow \infty$.
\end{itemize}
\end{itemize}

\section*{4 Optimal coordinates for conifolds}

\subsection*{4.1 Analysis on conifolds}

In order to do analysis of partial differential equations on conifolds, one has to work with weighted function spaces. In the following we recall some well-known facts for weighted Sobolev and Hölder spaces and Laplace type operators on conifolds which can be found in many articles and textbooks, see e.g. [Bar86, LM85, Mel93, Pac13].

Definition 4.1. Let $(M, g)$ be a conifold. A smooth function $\rho: M \rightarrow(0, \infty)$ is called a radius function if at each end $M_{i}$, there exists an asymptotic chart $\varphi_{i}$ such that $\left(\varphi_{i}\right)_{*} \rho=r$.

For simplicity, we assume that $M$ has only one end $M \backslash K$ which is either asymptotically conical or conically singular. Moreover, for fixed $R \in \mathbb{R}$, we will use the notation
$$M_{<R}:=\{p \in M \mid \rho(p)<R\},$$
and similarly with $M_{>R}:=\{p \in M \mid \rho(p)>R\}$. On a cone $\bar{M}$, we pick the function $r$ as radius function and use the notation $\bar{M}_{<R}$ etc. accordingly (cf. Remark 4.2)

For $p \in[1, \infty)$ and a weight $\beta \in \mathbb{R}$, we define the space $L_{\beta}^{2}(M)$ as the closure of $C_{\mathrm{cs}}^{\infty}(M)$ with respect to the norm
$$\|u\|_{L_{\beta}^{2}}=\left(\int_{M}\left|\rho^{-\beta} u\right|^{2} \rho^{-n} d \mu\right)^{1 / 2},$$
and the weighted Sobolev spaces $H_{\beta}^{k}(M)$ as the closure of $C_{\mathrm{cs}}^{\infty}(M)$ under
$$\|u\|_{H_{\beta}^{k}}=\sum_{l=0}^{k}\left\|\nabla^{l} u\right\|_{L_{\beta-l}^{2}} .$$
The weighted Hölder spaces are defined as the set of maps $u \in C_{\text {loc }}^{k, \alpha}(M), \alpha \in(0,1)$ such that the norm
$$\begin{aligned}
\|u\|_{C_{\beta}^{k, \alpha}}=\sum_{l=0}^{k} & \sup _{x \in M} \rho^{-\beta+l}(x)\left|\nabla^{l} u(x)\right| \\
& \quad+\sup _{\substack{x, y \in M \\
0<d(x, y)<\operatorname{inj}(M)}} \min \left\{\rho^{-\beta+k+\alpha}(x), \rho^{-\beta+k+\alpha}(y)\right\} \frac{\left|\tau_{x}^{y} \nabla^{k} u(x)-\nabla^{k} u(y)\right|}{d(x, y)^{\alpha}},
\end{aligned}$$
is finite. Here $\tau_{x}^{y}$ denotes the parallel transport from $x$ to $y$ along the shortest geodesic joining $x$ and $y$, and $d(x, y)$ is the Riemannian distance between $x$ and $y$. All these spaces are Banach spaces, the spaces $H_{\beta}^{k}(M)$ are Hilbert spaces and their underlying topological vector space structures do not depend on the choice of the radius function $\rho$. All these definitions extend to Riemannian vector bundles with a metric connection in an obvious manner [Pac13]. In the literature, there are different notational conventions for weighted spaces. We follow the more standard convention used in [Bar86, Pac13]. We have, for every $\beta \in \mathbb{R}, k, l \in \mathbb{N}_{0}$ and $\alpha \in(0,1)$, the obvious embeddings
$$H_{\beta}^{k+l} \subset H_{\beta}^{k}, \quad C_{\beta}^{k+1, \alpha} \subset C_{\beta}^{k, \alpha} .$$
In the asymptotically conical case, we further have the embeddings
$$H_{\beta}^{k} \subset H_{\beta^{\prime}}^{k}, \quad C_{\beta}^{k, \alpha} \subset C_{\beta^{\prime}}^{k, \alpha}, \quad C_{\beta}^{k, \alpha} \subset H_{\beta^{\prime}}^{k}$$
for any $\beta<\beta^{\prime}$, the Sobolev embedding
$$H_{\beta}^{k+l} \subset C_{\beta^{\prime}}^{k, \alpha}$$
for $l>n / 2$ and $\beta<\beta^{\prime}$, and pointwise multiplication extends to a continuous map
$$\cdot: H_{\beta}^{k} \times H_{\beta}^{k} \rightarrow H_{\beta}^{k}$$
for $\beta \leq 0$. Conversely, in the conically singular case, we have
$$H_{\beta}^{k} \subset H_{\beta^{\prime}}^{k}, \quad C_{\beta}^{k, \alpha} \subset C_{\beta^{\prime}}^{k, \alpha}, \quad C_{\beta}^{k, \alpha} \subset H_{\beta^{\prime}}^{k}$$
for any $\beta^{\prime}<\beta$, the Sobolev embedding
$$H_{\beta}^{k+l} \subset C_{\beta^{\prime}}^{k, \alpha}$$
for $l>n / 2$ and $\beta^{\prime}<\beta$, and pointwise multiplication extends to a continuous map
$$\cdot: H_{\beta}^{k} \times H_{\beta}^{k} \rightarrow H_{\beta}^{k}$$
for $\beta \geq 0$.
Remark 4.2. If $M$ is a conifold with multiple ends which might be AC as well as CS, these function spaces are generalized as follows: We index the AC ("large") ends by $\{1, \ldots, a\}$ and the CS ("small") ends by $\{1, \ldots, b\}$. Consider the vector $\beta=(\zeta, \sigma)=\left(\zeta_{1}, \ldots, \zeta_{a}, \sigma_{1}, \ldots, \sigma_{b}\right) \in \mathbb{R}^{a+b}$, where each $\zeta_{i}$ refers to an AC end and each $\sigma_{j}$ refers to a CS end. We write $\zeta \leq \zeta^{\prime}$ (resp. $\geq,<,>$ ) whenever $\zeta_{i} \leq \zeta_{i}^{\prime}$ (resp. $\geq,<,>)$ for each $i$, and similarly for $\sigma$. For $i \in \mathbb{Z}$, we write $\beta+i:=(\zeta+i, \sigma+i):=\left(\zeta_{1}+i, \ldots, \zeta_{a}+\right.$ $\left.i, \sigma_{1}+i, \ldots, \sigma_{b}+i\right)$. Given $\beta=(\zeta, \sigma) \in \mathbb{R}^{a+b}$, choose a smooth function on $M$ (again denoted by $\beta: M \rightarrow \mathbb{R}$ ) which at each AC end $M_{i}$ coincides with $\zeta_{i}$ and at each CS end $M_{j}$ coincides with $\sigma_{j}$. Then we can define the weighted Sobolev spaces $H_{\beta}^{k}=H_{\zeta, \sigma}^{k}$ and the weighted Hölder spaces $C_{\beta}^{k, \alpha}=C_{\zeta, \sigma}^{k, \alpha}$ exactly as above. The embedding properties from above hold for $\beta=(\zeta, \sigma), \beta^{\prime}=\left(\zeta^{\prime}, \sigma^{\prime}\right) \in \mathbb{R}^{a+b}$ with $\zeta<\zeta^{\prime}$ and $\sigma^{\prime}<\sigma$ and the multiplication properties for $\zeta \leq 0$ and $\sigma \geq 0$.

Definition 4.3. A Laplace type operator $\Delta_{V}$ on a Riemannian vector bundle $V$ with connection over an asymptotically conical/conically singular manifold $M$ is called asymptotically conical/conically singular of order $\alpha$ if there exists a Riemannian vector bundle $\bar{V}$ with connection over $(R, \infty) \times \widehat{M}$ (resp. $(0, \epsilon) \times$ $\widehat{M}$ ), a conical operator $\bar{\Delta}_{\bar{V}}$, a vector bundle isomorphism $\Phi:\left.V\right|_{M \backslash K} \rightarrow \bar{V}$ covering an asymptotic chart $\phi$ and a constant $\alpha>0$, such that for any section $u=\mathcal{O}_{\infty}\left(r^{\beta}\right), \beta \in \mathbb{R}$, we have
$$\bar{\nabla}^{k}\left(\bar{\Delta}_{\bar{V}} \Phi(u)-\Phi\left(\Delta_{V} u\right)\right)=\mathcal{O}_{\infty}\left(r^{\beta-\alpha-2-k}\right) \quad \text { for all } k \in \mathbb{N}_{0}$$
as $r \rightarrow \infty$ in the AC case, and
$$\bar{\nabla}^{k}\left(\bar{\Delta}_{\bar{V}} \Phi(u)-\Phi\left(\Delta_{V} u\right)\right)=\mathcal{O}_{\infty}\left(r^{\beta+\alpha-2-k}\right) \quad \text { for all } k \in \mathbb{N}_{0}$$
as $r \rightarrow 0$ in the CS case. A Laplace type operator over a conifold is called a conifold operator if it is asymptotically conical and conically singular at each asymptotically conical and conically singular end, respectively.

For any $\beta \in \mathbb{R}$, a conifold operator $\Delta_{V}$ defines a continuous map
$$\Delta_{V}: W_{\beta}^{k, p}(V) \rightarrow W_{\beta-2}^{k-2, p}(V) .$$

Additionally, it is Fredholm for all $\beta \in \mathbb{R}$ up to a discrete set, see e.g. [Mel93, Proposition 5.64]: Let
$$D=\left\{\operatorname{Re}\left(\xi_{ \pm}(\nu)\right) \mid \nu \in \operatorname{spec}\left(\square_{V}\right)\right\},$$
where $\square_{V}$ is the tangential operator of the conical operator $\bar{\Delta}_{\bar{V}}$. In other words $D$ is the set of the real part of the indicial roots of $\bar{\Delta}_{\bar{V}}$. We call $\beta$ nonexceptional whenever $\beta \in \mathbb{R} \backslash D$ and exceptional, whenever $\beta \in D$. It is a standard fact that the operator (4.2) is Fredholm if and only if $\beta$ is nonexceptional. An analogous statement holds for weighted Hölder spaces but we will not need it for our purposes.

To conclude this short introduction, we mention an important duality argument. Note that the $L^{2}$ -scalar product induces a bounded map
$$(., .): L_{\beta}^{2} \times L_{-n-\beta}^{2} \rightarrow \mathbb{R}$$
consequently we can think of $L_{-n-\beta}^{2}$ as being the dual space of $L_{\beta}^{2}$. Motivated by this, we define
$$H_{\beta}^{-k}:=\left(H_{-n-\beta}^{k}\right)^{\prime}, \quad k \in \mathbb{N} .$$
If $\Delta_{V}$ is formally self-adjoint, the dual map of
$$\Delta_{V}: H_{\beta}^{k}(V) \rightarrow W_{\beta-2}^{k-2}(V)$$
is, via the $L^{2}$-pairing, again given by $\Delta_{V}$, seen as a map
$$\Delta_{V}: H_{2-n-\beta}^{2-k}(V) \rightarrow H_{-n-\beta}^{-k}(V) .$$
Recall that a Fredholm map is surjective if its dual map is injective. Therefore,
$$\Delta_{V}: H_{\beta}^{k}(V) \rightarrow H_{\beta-2}^{k-2}(V)$$
is surjective if and only if
$$\operatorname{ker}_{H_{2-n-\beta}^{2-k, q}}\left(\Delta_{V}\right)=0$$
However, by elliptic regularity for weighted spaces,
$$\operatorname{ker}_{H_{2-n-\beta}^{2-k}}\left(\Delta_{V}\right)=\operatorname{ker}_{H_{2-n-\beta}^{l}}\left(\Delta_{V}\right)$$
for every $l \in \mathbb{N}$. This gives us a useful criterion for checking isomorphism properties for $\Delta_{V}$.
For example, consider the conical operator $\bar{\Delta}_{\bar{V}}$ over a Riemannian cone $(\bar{M}, \bar{g})$. In view of Remark 4.2, we may think of $\bar{M}$ as a conifold with one AC and one CS end and choose the same weight $\beta$ on both ends. We may choose $\rho=r$ as the radius function. By (3.6), an element $u \in \operatorname{ker}\left(\bar{\Delta}_{\bar{V}}\right)$ defined on all of $\bar{M}$ can not satisfy $u \in o\left(r^{\alpha}\right)$ both as $r \rightarrow 0$ and $r \rightarrow \infty$ for any $\alpha \in \mathbb{R}$. For this reason,
$$\bar{\Delta}_{\bar{V}}: H_{\beta}^{k}(\bar{V}) \rightarrow H_{\beta-2}^{k-2}(\bar{V})$$
is injective for all weights $\beta$. By the duality argument above, $\bar{\Delta}_{\bar{V}}$ is also surjective for all nonexceptional weights $\beta$. Summing up, we conclude:

Proposition 4.4. A conical operator $\bar{\Delta}_{\bar{V}}$ over a Riemannian cone $(\bar{M}, \bar{g})$, seen as an operator
$$\bar{\Delta}_{\bar{V}}: H_{\beta}^{k}(V) \rightarrow H_{\beta-2}^{k-2}(V)$$
is an isomorphism for every nonexceptional weight $\beta$.

\subsection*{4.2 Decay of Ricci-flat metrics on the cone}

Recall that for two different Riemannian metrics $g, \tilde{g}$ the vector field $V(g, \tilde{g})$ is given in local coordinates by
$$V(g, \tilde{g})^{l}:=g^{i j}\left(\Gamma(g)_{i j}^{l}-\Gamma(\tilde{g})_{i j}^{l}\right) .$$

Definition 4.5. We say that a metric $g$ is in Bianchi gauge with respect to $\tilde{g}$ if $V(g, \tilde{g})=0$.
Theorem 4.6. Let $(\bar{M}, \bar{g})$ be a Ricci-flat cone and let $g$ be a Ricci-flat metric defined on an open set $U \subset \bar{M}$ which is in Bianchi gauge with respect to $\bar{g}$. Then the following assertions hold:
\begin{itemize}
\item[(i)] If $U$ is an open neighborhood of 0 (i.e. if $U$ contains $\bar{M}_{<R}$ for some $R \in \mathbb{R}$ ) and $g-\bar{g}=\mathcal{O}_{2}\left(r^{\alpha}\right)$ for some $\alpha>0$ as $r \rightarrow 0$, then we have $g-\bar{g}=\mathcal{O}_{\infty}\left(r^{\xi_{+}^{B}}\right)$ as $r \rightarrow 0$.
\item[(ii)] If $U$ is an open neighborhood of $\infty$ (i.e. if $M \backslash U$ is compact) and $g-\bar{g} \in \mathcal{O}_{2}\left(r^{-\alpha}\right)$ for some $\alpha>0$ as $r \rightarrow \infty$, then we divide into two subcases:
\begin{itemize}
\item[] (iia) If $\operatorname{spec}\left(\square_{L}\right) \cap\left[-\frac{(n-2)^{2}}{4}, 0\right) \neq\left\{-\frac{(n-2)^{2}}{4}\right\}$, then $g-\bar{g}=\mathcal{O}_{\infty}\left(r^{-\xi_{-}^{B}}\right)$ as $r \rightarrow \infty$.
\item[] (iib) If $\operatorname{spec}\left(\square_{L}\right) \cap\left[-\frac{(n-2)^{2}}{4}, 0\right)=\left\{-\frac{(n-2)^{2}}{4}\right\}$, then $g-\bar{g}=\mathcal{O}_{\infty}\left(r^{-\frac{n-2}{2}} \log (r)\right)$ as $r \rightarrow \infty$.
\end{itemize}
\end{itemize}
Proof. In this proof, all norms and tensor products are taken with respect to $\bar{g}$. By the assumptions of the theorem, we have the equation
$$2 \operatorname{Ric}_{g}=\mathcal{L}_{V(g, \bar{g})} \bar{g}$$
on $U$, which can be written with respect to the difference $h=g-\bar{g}$ as
$$\bar{\Delta}_{L} h=g^{-1} * \overline{\mathrm{Rm}} * h * h+g^{-1} * g^{-1} * \bar{\nabla} h * \bar{\nabla} h+g^{-1} * \bar{\nabla}^{2} h * h,$$
where, as usual, * denotes finite linear combinations of tensor contractions with covariantly constant coefficients. This follows essentially from [Shi89, Lemma 2.1], but is carried out in more detail in [KP20, Lemma 3.1]. From here on, the proof is a standard iteration procedure in weighted function spaces, but we decided to present it here for completeness. We focus on case (i), the other one is completely analogous. Without loss of generality, we assume that $h$ is defined on $\bar{M}_{<R}$ for some $R>0$. We extend $h$ smoothly to a tensor $\bar{h}$ on $\bar{M}$ such that
$$\bar{h} \equiv 0 \text { on } \bar{M}_{>2 R}, \quad \bar{h} \equiv h \text { on } \bar{M}_{<R} .$$
At first, by elliptic regularity, $\bar{h}=\mathcal{O}_{\infty}\left(r^{\alpha}\right)$ as $r \rightarrow 0$ (and for trivial reasons also as $r \rightarrow \infty$ ). Thus by (4.3),
$$\bar{\Delta}_{L} \bar{h}=\mathcal{O}_{\infty}\left(r^{2 \alpha-2}\right),$$
in both the cases $r \rightarrow \infty$ and $r \rightarrow 0$. Therefore,
$$\bar{\Delta}_{L} \bar{h} \in H_{2 \alpha-2-\epsilon}^{k}\left(S^{2} M\right)$$
for any $k \in \mathbb{N}_{0}$ and for any $\epsilon>0$ and by Proposition 4.4. we find a tensor $h_{1} \in H_{-2 \alpha-\epsilon}^{k+2}\left(S^{2} M\right)$ such that
$$\bar{\Delta}_{L} h_{1}=\bar{\Delta}_{L} \bar{h},$$
provided that $2 \alpha-\epsilon$ is a nonexceptional weight. Note that $h_{1}$ is independent of the choice of $k$ (but not of the choice of $\epsilon$ ) as $H_{2 \alpha-\epsilon}^{k+2} \subset H_{2 \alpha+\epsilon}^{k^{\prime}+2}$ if $k \leq k^{\prime}$. In particular, $h_{1} \in H_{2 \alpha-\epsilon}^{k}$ for all $k \in \mathbb{N}$ and $\epsilon>0$ so by Sobolev embedding, $h_{1} \in \mathcal{O}_{\infty}\left(r^{2 \alpha-2 \epsilon}\right)$.

Because $h_{0}:=\bar{h}-h_{1} \in \operatorname{ker}\left(\Delta_{L}\right)$ and $h_{0} \in \mathcal{O}_{\infty}\left(r^{\beta}\right)$ as $r \rightarrow 0$, we know by Corollary 3.9 that $h_{1} \in \mathcal{O}_{\infty}\left(r^{\xi_{+}}\right)$. We obtain on the set $\bar{M}_{<R}$ that
$$h=\bar{h}=h_{0}+h_{1} \in \mathcal{O}_{\infty}\left(r^{\xi_{+}}\right)+\mathcal{O}_{\infty}\left(r^{2 \alpha-2 \epsilon}\right) .$$
as $r \rightarrow 0$. If $2 \alpha-2 \epsilon<\xi_{+}$for some $\epsilon>0$, we are done. Otherwise we repeat the same procedure again, starting with the new decay rate $\alpha^{\prime}:=2 \alpha-2 \epsilon$. After iterating this procedure at most a finite number of times, we will be in the situation where $2 \alpha-2 \epsilon<\xi_{+}$. In this case, we can now conclude
$$h=\bar{h}=h_{0}+h_{1} \in \mathcal{O}_{\infty}\left(r^{\xi_{+}}\right),$$
where $\bar{\Delta}_{L} h_{0}=0$ and $h_{1} \in \mathcal{O}_{\infty}\left(r^{\xi_{+}+\epsilon}\right)$ for some $\epsilon>0$. The equation $V(g, \bar{g})=0$ is equivalent to
$$\begin{aligned}
0=\frac{1}{2} g^{i j}\left(\bar{\nabla}_{i} h_{j k}+\bar{\nabla}_{j} h_{i k}-\bar{\nabla}_{k} h_{i j}\right) & =B_{\bar{g}}(h)+\frac{1}{2}\left(g^{i j}-\bar{g}^{i j}\right)\left(\bar{\nabla}_{i} h_{j k}+\bar{\nabla}_{j} h_{i k}-\bar{\nabla}_{k} h_{i j}\right) \\
& =B_{\bar{g}}\left(h_{0}\right)+B_{\bar{g}}\left(h_{1}\right)-h_{l m} g^{i l} \bar{g}^{j m}\left(\bar{\nabla}_{i} h_{j k}+\bar{\nabla}_{j} h_{i k}-\bar{\nabla}_{k} h_{i j}\right) \\
& =B_{\bar{g}}\left(h_{0}\right)+\mathcal{O}\left(r^{\xi_{+}-1+\epsilon}\right),
\end{aligned}$$
and because $B_{\bar{g}}\left(h_{0}\right) \in \mathcal{O}_{\infty}\left(r^{\xi_{+}-1}\right)$ we conclude $B_{\bar{g}}\left(h_{0}\right)=0$. Thus, if $\xi_{+}<\xi_{+}^{B}$, we get $h_{0}=0$ and therefore $h \in \mathcal{O}_{\infty}\left(r^{\xi_{+}+\epsilon}\right)$. We then continue with the same procedure as above till we arrive at a decomposition $h=h_{0}+h_{1}$ with $\Delta_{L} h_{0}=0$ and $B_{\bar{g}}\left(h_{0}\right)=0$, where $h_{1}$ decays faster than $h_{0}$. By Corollary 3.21, we know that $h_{0}=\mathcal{O}_{\infty}\left(r^{\xi_{+}^{B}}\right)$ which implies the desired result. \(\square\)

\subsection*{4.3 The Bianchi gauge}

In this subsection, we want to find out under which conditions the Bianchi gauge condition actually defines a reasonable gauge. For simplicity, we assume again that our manifold has only one end which is either asymptotically conical or conically singular. We start with the following observation:

Lemma 4.7. The set
$$\mathcal{G}_{\beta}^{k}:=\left\{\tilde{g} \in H_{\beta}^{k}\left(S_{+}^{2} M\right) \mid V(g, \tilde{g})=0\right\},$$
is an open subset of a vector space. In particular, it is a submanifold of $H_{\beta}^{k}\left(S_{+}^{2} M\right)$.
Proof. Choose a point $p \in M$, and $h=\tilde{g}-g$. In $g$-normal coordinates around $p$, we compute at $p$ that
$$\begin{aligned}
V(g, \tilde{g})^{k}=-g^{i j} \Gamma(\tilde{g})_{i j}^{k} & =-\frac{1}{2} g^{i j} \tilde{g}^{k l}\left(\partial_{i} \tilde{g}_{l j}+\partial_{j} \tilde{g}_{l i}-\partial_{l} \tilde{g}_{i j}\right) \\
& =-\frac{1}{2} g^{i j} \tilde{g}^{k l}\left(\partial_{i} h_{l j}+\partial_{j} h_{l i}-\partial_{l} h_{i j}\right) \\
& =-\frac{1}{2} g^{i j} \tilde{g}^{k l}\left(\nabla_{i} h_{l j}+\nabla_{j} h_{l i}-\nabla_{l} h_{i j}\right)=\tilde{g}^{k l} B_{g}(h)_{l} .
\end{aligned}$$
Therefore,
$$\mathcal{G}_{\beta}^{k}=\left\{g+h \in H_{\beta}^{k}\left(S_{+}^{2} M\right) \mid B_{g}(h)=0\right\}=\left\{h \in H_{\beta}^{k}\left(S_{+}^{2} M\right) \mid B_{g}(h)=0\right\}$$
since $B_{g}(g)=0$. Hence, we have redefined $\mathcal{G}_{\beta}^{k}$ by a linear equation, which proves the lemma. \(\square\)

Remark 4.8. The fact that the above gauge condition is linear is the reason why we prefer it over the condition $V(\tilde{g}, g)=0$ for fixed $g$. In fact, if $g$ has nontrivial Killing fields, the differential of the linearization is not surjective and the set of metrics satisfying this condition may fail to form a manifold.

Lemma 4.9. The connection Laplacian
$$\Delta_{1}: H_{\beta}^{k+2}\left(T^{*} M\right) \rightarrow H_{\beta-2}^{k}\left(T^{*} M\right)$$
is injective for nonexceptional $\beta<0$ and surjective for every nonexceptional $\beta>2-n$.
Proof. If $\omega \in H_{\beta}^{k+2}(T M)$ satisfies $\Delta_{1} \omega=0$, then
$$\Delta|\omega|^{2}+2|\nabla \omega|^{2}=0 .$$
By the maximum principle, we conclude that any bounded harmonic vector field on $M$ is parallel. In particular, its pointwise norm is constant and
$$\Delta_{1}: H_{\beta}^{k+2}\left(T^{*} M\right) \rightarrow H_{\beta-2}^{k}\left(T^{*} M\right)$$
is an injective Fredholm operator for every nonexceptional $\beta<0$. The surjectivity for $\beta>2-n$ follows from duality via the $L^{2}$-pairing, as explained in Subsection 4.1 \(\square\)

Lemma 4.10. Pick $\beta>0$ such that the two operators
$$B: H_{\beta}^{k}\left(S^{2} M\right) \rightarrow H_{\beta-1}^{k-1}\left(T^{*} M\right)$$
and
$$\Delta_{1}: H_{\beta+1}^{k+1}\left(T^{*} M\right) \rightarrow H_{\beta-1}^{k-1}\left(T^{*} M\right)$$
are both Fredholm. Then, we have $B h \in \Delta_{1}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)$ if and only if
$$\left(h, B^{*} \eta\right)_{L^{2}}=0$$
for every $\eta \in H_{1-n-\beta}^{k}\left(T^{*} M\right)$ with $\Delta_{1} \eta=0$. Here, $B^{*} \omega=\delta^{*} \omega+\frac{1}{2} \delta \omega \cdot g$ is the formal adjoint of $B$.
Proof. By duality, $B h=\Delta_{1} \omega$ for some $\omega \in H_{\beta+1}^{k+1}\left(T^{*} M\right)$ if and only if
$$(B h, \eta)_{L^{2}}=0$$
for all $\eta \in H_{1-n-\beta}^{1-k}\left(T^{*} M\right)$ with $\Delta_{1} \eta=0$. The criterion (4.5) now follows from integration by parts. By elliptic regularity, $\eta \in H_{1-n-\beta}^{k}\left(T^{*} M\right)$. \(\square\)

Lemma 4.11. Let $\beta \in \mathbb{R}$ be as in Lemma 4.10 Then there exist a number $N \in \mathbb{N}$ and for any open precompact subset $U \subset M$ a set of tensors $h_{i}, i=1, \ldots N$, with support in $\bar{U}$ such that
$$H_{\beta}^{k}\left(S^{2} M\right)=B^{-1}\left(\Delta_{1}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)\right) \oplus\left\{\sum_{i=1}^{N} \alpha_{i} h_{i} \mid \alpha_{i} \in \mathbb{R}\right\} .$$
Proof. Note that on a Ricci-flat manifold, $2 \delta \circ B^{*}=\Delta_{1}$, therefore $\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(B^{*}\right) \subset \operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)$. Since $\operatorname{ker}_{H_{\beta+1}^{k+1}} \Delta_{1}$ is finite dimensional, we can find a finite-dimensional vector space $V$ such that
$$\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)=\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(B^{*}\right) \oplus V .$$
By the commutation formulas (2.11), $B^{*} \eta \in \operatorname{ker}\left(\Delta_{L}\right)$ for $\eta \in V$. By assumption, $B^{*} \eta \neq 0$ if $\eta \neq 0$. In particular, $B^{*} \eta$ does not vanish identically on any open subset of $M$ by elliptic theory. For a given open
and precompact set $U \subset M$, we choose a bump function $\chi: M \rightarrow[0,1]$ which is strictly positive in $U$ and vanishes identically on $M \backslash U$. Then the bilinear form
$$A: V \times V \rightarrow \mathbb{R}, \quad\left(\eta_{1}, \eta_{2}\right) \mapsto \int_{M} \chi\left\langle B^{*} \eta_{1}, B^{*} \eta_{2}\right\rangle d V$$
is an inner product because the tensors $B^{*} \eta_{i}$ do not vanish identically on any open subset of $U$. Therefore, we may choose a basis $\left\{\eta_{i} \mid i=1, \ldots N\right\}$ of $V$ which is orthonormal with respect to the inner product $A$. Define $h_{i}:=\chi B^{*} \eta_{i}$ for $i \in\{1, \ldots, N\}$. Now let $h \in H_{\beta}^{k}\left(S^{2} M\right)$ be arbitrary and make the ansatz
$$\hat{h}=h-\sum_{i=1}^{N} \alpha_{i} h_{i}, \quad \alpha_{i} \in \mathbb{R} .$$
By Lemma 4.10, we have
$$\hat{h} \in B^{-1}\left(\Delta_{1}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)\right)$$
if and only if $\left(\hat{h}, B^{*} \eta\right)_{L^{2}}=0$ for all $\eta \in V$, or equivalently, if and only if $\left(\hat{h}, B^{*} \eta_{j}\right)_{L^{2}}=0$ for all $j \in\{1, \ldots, N\}$. By construction of the $h_{i}$,
$$\left(\hat{h}, B^{*} \eta_{j}\right)_{L^{2}}=\left(h, B^{*} \eta_{j}\right)_{L^{2}}-\sum_{i=1}^{N} \alpha_{i}\left(h_{i}, B^{*} \eta_{j}\right)_{L^{2}}=\left(h, B^{*} \eta_{j}\right)_{L^{2}}-\sum_{i=1}^{N} \alpha_{i} A\left(\eta_{i}, \eta_{j}\right) .$$
Because $A\left(\eta_{i}, \eta_{j}\right)=\delta_{i j}$ by construction, (4.6) is satisfied if and only if $\alpha_{i}=\left(\hat{h}, B^{*} \eta_{i}\right)_{L^{2}}$ for all $i \in$ $\{1, \ldots, N\}$. This proves the Lemma. \(\square\)

Lemma 4.12. We have
$$\operatorname{ker}_{H_{\beta}^{k}}(B) \cap \delta^{*}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)=\delta^{*}\left(\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)\right) .$$
Furthermore, for any subspace $Z_{\beta}^{k} \subset \operatorname{ker}_{H_{\beta}^{k}}(B)$ with
$$\operatorname{ker}_{H_{\beta}^{k}}(B)=Z_{\beta}^{k} \oplus \delta^{*}\left(\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)\right),$$
we also have
$$B^{-1}\left(\Delta_{1}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)\right)=Z_{\beta}^{k} \oplus \delta^{*}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right) .$$
Proof. The first assertion is immediate from the formula
$$B \circ \delta^{*}=\Delta_{1}+\operatorname{Ric}=\Delta_{1},$$
which follows from a straightforward calculation. Since $\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)$ is finite dimensional we can choose a subspace $Z_{\beta}^{k} \subset \operatorname{ker}_{H_{\beta}^{k}}(B)$ such that
$$\operatorname{ker}_{H_{\beta}^{k}}(B)=Z_{\beta}^{k} \oplus \delta^{*}\left(\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)\right) .$$
For the second assertion, consider the sum $Z_{\beta}^{k}+\delta^{*}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)$. Due to the first assertion, we have
$$Z_{\beta}^{k} \cap \delta^{*}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)=0,$$
so the sum is direct. It remains to show that the sum is equal to $B^{-1}\left(\Delta_{1}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)\right)$. For an arbitrary $h \in B^{-1}\left(\Delta_{1}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right)\right)$, we choose a form $\omega_{0} \in H_{\beta+1}^{k+1}\left(T^{*} M\right)$ such that $B h=\Delta_{1} \omega_{0}$ so that $h-\delta^{*} \omega_{0} \in \operatorname{ker}_{H_{\beta}^{k}}(B)$. Now by the first assertion, we can write $h-\delta^{*} \omega_{0}=\delta^{*} \omega_{1}+h_{1}$, with some $h_{1} \in Z_{\beta}^{k}$ and $h=\delta^{*}\left(\omega_{0}+\omega_{1}\right)+h_{1}$ is the desired decomposition. \(\square\)

The following proposition asserts that the Bianchi gauge is a very reasonable one for any asymptotically conical Ricci-flat manifold $(M, g)$ : Any metric $\tilde{g}$ sufficiently close to $g$ can be brought by a diffeomorphism into Bianchi gauge, possibly up to an open subset we are free to choose.

Theorem 4.13. Let ( $M^{n}, g$ ) be an AC Ricci-flat manifold and let $k>n / 2+1$ and $\beta<0$ be such that $\beta+1$ is a nonexceptional value for $\Delta_{1}$ and $\beta$ is a nonexceptional value for $B$. Pick an open and precompact subset $U \subset M$ and a complement $Z_{\beta}^{k}$ as (4.7). Set
$$\mathcal{H}_{\beta}^{k}:=\left\{g+h \mid h \in Z_{\beta}^{k}\right\} .$$
Then there exists an $H_{\beta}^{k}$-neighborhood $\mathcal{U}_{\beta}^{k}$ of $g$ in the space of metrics such that for any $\tilde{g} \in \mathcal{U}_{\beta}^{k}$, there exists a diffeomorphism $\varphi$ which is $H_{\beta+1}^{k+1}$-close to the identity and a tensor $\tilde{h}$ with $\operatorname{supp}(\tilde{h}) \subset \bar{U}$ such that $\varphi^{*}(\tilde{g}+\tilde{h}) \in \mathcal{H}_{\beta}^{k}$.

Remark 4.14. On the other hand, if $\beta>1-n$, the assertion holds without adding the tensor $k$, because $\Delta_{1}$ is then surjective on $H_{\beta+1}^{k+1}\left(T^{*} M\right)$, see again Lemma 4.9

In case of multiple ends, we also have an analogous assertion. In this case one would work with function spaces with multiple weights which were briefly introduced in Remark 4.2 One would then choose a tuple $\beta=(\zeta, \sigma)$ with $\zeta<0$ and $\sigma>0$.

Proof. Clearly, we have
$$T_{g} \mathcal{H}_{\beta}^{k}=Z_{\beta}^{k} .$$
Choose tensors $h_{i}, i \in\{1, \ldots, N\}$ with support in $\bar{U}$ as in Lemma 4.11 and let $\widetilde{V}=\operatorname{span}_{\mathbb{R}}\left(h_{i}\right)_{1 \leq i \leq N}$. Then, Lemma 4.12 implies that
$$H_{\beta}^{k}\left(S^{2} M\right)=Z_{\beta}^{k} \oplus \delta^{*}\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right) \oplus \widetilde{V}=T_{g} \mathcal{H}_{\beta}^{k} \oplus \mathcal{L} \circ \sharp\left(H_{\beta+1}^{k+1}\left(T^{*} M\right)\right) \oplus \widetilde{V} .$$
Here, we used that $2 \delta^{*}=\mathcal{L} \circ \sharp$, where $\mathcal{L}: X \rightarrow \mathcal{L}_{X} g$ is the Lie derivative and $\sharp: \omega \mapsto \omega^{\sharp}$ is the sharp operator with respect to $g$. Note that because $\beta+1<1$, the vector fields in $H_{\beta+1}^{k+1}(T M)$ are all complete, because they grow slower than linearly. Therefore, we have a well-defined map
$$H_{\beta+1}^{k+1}(T M) \ni X \mapsto \psi_{X} \in H_{\beta}^{k}(\operatorname{Diff}(M)),$$
where $\psi_{X}$ is the flow of $X$, evaluated at time $t=1$. Now we consider the smooth map
$$\Psi: \mathcal{H}_{\beta}^{k} \times H_{\beta+1}^{k+1}(T M) \times \widetilde{V} \rightarrow H_{\beta}^{k}\left(S_{+}^{2} M\right), \quad(g, X, \tilde{h}) \mapsto\left(\psi_{X}\right)^{*} g+\tilde{h} .$$
Its differential at $(g, 0,0)$ corresponds to the decomposition (4.8). Therefore, $\Psi$ is a local diffeomorphism from a neighborhood of $(g, 0,0)$ onto a neighborhood of $g$ by the implicit function theorem. This proves the Proposition. \(\square\)

Remark 4.16. Note that the essential reason for the assumption $\beta<0$ is to guarantee completeness for the vector fields in $H_{\beta+1}^{k+1}$. In the conically singular case, completeness is guaranteed by assuming $\beta>0$ as this implies that the vector fields decay at the singularity like $\mathcal{O}\left(r^{1+\epsilon}\right)$.

This section is devoted to the proof of Theorem 1.7 and essentially builds up on the slice theorem 4.13. We prove only for the case of one AC end. The case of one conically singular end is completely analogous. The proof for multiple ends of both types is also analogous and based on a slice theorem for multiple ends, see Remark 4.15. The details are left to the reader. Recall the notation $\bar{M}_{<R}$ and $\bar{M}_{>R}$ from (4.1).

Theorem 4.17. Let $(M, g)$ be an asymptotically conical Ricci-flat manifold. Then there exist compact set $K \subset M$ and an asymptotic chart $\varphi: M \backslash K \rightarrow \bar{M}_{>R}$ such that
\begin{itemize}
\item[(i)] if $(\bar{M}, \bar{g})$ is not resonance-dominated, we have $\varphi_{*} g-\bar{g} \in \mathcal{O}_{\infty}\left(r^{-\xi-}\right)$ as $r \rightarrow \infty$ where $\xi_{-}$has been defined in (1.2),
\item[(ii)] if $(\bar{M}, \bar{g})$ is resonance-dominated, we have $\varphi_{*} g-\bar{g} \in \mathcal{O}_{\infty}\left(r^{-\frac{n-2}{2}} \log (r)\right)$ as $r \rightarrow \infty$.
\end{itemize}

Proof. Because ( $M, g$ ) is asymptotically conical, we find a compact set $K_{0} \subset M$ and a diffeomorphism $\varphi_{0}: M \backslash K_{0} \rightarrow \bar{M}_{>R}$ such that $\left(\varphi_{0}\right)^{*} g-\bar{g} \in \mathcal{O}_{\infty}\left(r^{-\tau}\right)$ for some $\tau>0$ as $r \rightarrow \infty$. Let $f_{R}: \mathbb{R}_{\geq 0} \rightarrow$ [0, 1] be a smooth cutoff function such that
$$\left.f_{R}\right|_{[0, R]} \equiv 1,\left.\quad f_{R}\right|_{[2 R, \infty)} \equiv 0, \quad\left|f_{R}^{(k)}\right| \leq C_{k} \cdot R^{-k} \text { for all } k \in \mathbb{N}$$
for some constants $C_{i} \in \mathbb{R}$. Choose a radius function $\rho$ on $M$ such that $\left(\varphi_{0}\right)_{*} \rho=r$ and let $F_{R}=f_{R} \circ \rho$. Let $R_{0}:=\max _{K} \rho$, and for $R \geq R_{0}$, define a metric $g_{R}$ on $M$ by
$$g_{R}=F_{R} \cdot g+\left(1-F_{R}\right)\left(\varphi_{0}\right)^{*} \bar{g} .$$
Inside $M_{<R}$, the metric $g_{R}$ agrees with $g$ and on $M_{>2 R}$, the metric $g_{R}$ agrees with $\left(\varphi_{0}\right)^{*} \bar{g}$. Moreover, by the choice of $F_{R}$, we have $g_{R} \rightarrow g$ in $C_{-\tau}^{k}\left(S^{2} M\right)$ for any $k \in \mathbb{N}$ as $R \rightarrow \infty$. Thus for any $\beta>-\tau$, we
also get $g_{R} \rightarrow g$ in $H_{\beta}^{k}\left(S^{2} M\right)$ for any $k \in \mathbb{N}$. Choose $k>n / 2+2$ pick an open set $U \subset K_{0} \subset M$. Then by Theorem 4.13, there exists for a sufficiently large value for $R$ (which we denote again by $R$ ), a diffeomorphism $\psi_{R}$ and a tensor $h_{R}$ with compact support in $\bar{U} \subset K_{0}$ such that such that
$$V\left(g,\left(\psi_{R}\right)^{*}\left(g_{R}+h_{R}\right)\right)=0 .$$
Consider now the compact set $K:=\left(\varphi_{R}\right)^{-1}\left(K_{0}\right)$ and the diffeomorphism
$$\varphi:=\varphi_{0} \circ \psi_{R}: M \backslash K \rightarrow \bar{M}_{>R} .$$
By construction, the function $F_{R}$ vanishes on the set $M_{>2 R} \subset M$. Furthermore, $h_{R}$ vanishes on the domain of $\varphi_{0}$. Therefore we have on the set $\bar{M}_{>2 R}$ that $\varphi_{*}\left(\psi_{R}\right)^{*} g_{R}=\varphi_{*}\left(\psi_{R}\right)^{*}\left(\varphi_{0}\right)^{*} \bar{g}=\bar{g}$ and consequently by (4.9)
$$-2 \operatorname{Ric}_{\varphi_{*} g}+\mathcal{L}_{V\left(\varphi_{*}, \bar{g}\right)}\left(\varphi_{*} g\right)=\mathcal{L}_{V\left(\varphi_{*} g, \varphi_{*}\left(\psi_{R}\right)^{*} g_{R_{1}}\right)}\left(\varphi_{*} g\right)=0$$
Because $k$ was chosen to satisfy $k>n / 2+2$, we have $\varphi_{*} g-\bar{g}=\mathcal{O}_{2}\left(r^{-\beta / 2}\right)$. From Theorem 4.6, we obtain that
$$\varphi_{*} g-\bar{g}=\mathcal{O}_{\infty}\left(r^{-\xi_{-}^{B}}\right) .$$
If $\xi_{-}^{B}=\xi_{-}$, we are done. If $\xi_{-}^{B}<\xi_{-}$, we know that the leading term of the expansion at infinity is a Lie derivative. In other words, we can decompose
$$\varphi_{*} g-\bar{g}=\varphi_{*}\left(g-\left(\psi_{R}\right)^{*} g_{R}\right)=\bar{\delta}^{*} \bar{\omega}+\hat{h}, \quad \hat{h}=\mathcal{O}_{\infty}\left(r^{-\xi_{-}^{B}-\epsilon}\right),$$
for some $\epsilon>0$ and $\bar{\omega}$. In the following, we are going to show that the leading gauge term $\bar{\delta}^{*} \bar{\omega}$ indeed vanishes. Consider the difference
$$h_{0}:=\left(\psi_{R}\right)^{*}\left(g_{R}+h_{R}\right)-g \in H_{\beta}^{k}\left(S^{2} M\right), \quad \text { for } \quad k>\frac{n}{2}+2, \quad \beta>-\xi_{-}^{B} .$$
Consider the 1-form $\bar{\omega}$ in (4.10) and extend the form $\varphi^{*} \bar{\omega}$ on $M \backslash K$ to a smooth 1-form $\omega_{1}$ on all of $M$. Then, we have that
$$h_{0}=\delta^{*} \omega_{1}+h_{1}, \quad h_{1}=\mathcal{O}_{\infty}\left(r^{-\xi_{-}^{B}-\epsilon}\right) .$$
for some $\epsilon>0$. Recall that from the proof of Proposition 4.13, we have $h_{0} \in Z_{\beta}^{k}$, where $Z_{\beta}^{k}$ is a space such that
$$\operatorname{ker}_{H_{\beta}^{k}}(B)=Z_{\beta}^{k} \oplus \delta^{*}\left(\operatorname{ker}_{H_{\beta+1}^{k+1}}\left(\Delta_{1}\right)\right) .$$
Let us proceed with the tensor $h_{1}$. At first we know
$$h_{1} \in H_{\beta^{\prime}}^{k}\left(S^{2} M\right), \quad \text { for } \quad k>\frac{n}{2}+2, \quad \beta^{\prime}>-\xi_{-}^{B}-\epsilon .$$
Now choose $\beta^{\prime} \in\left(\xi_{-}^{B}-\epsilon, \xi_{-}^{B}\right)$ and choose a subspace $Z_{\beta^{\prime}}^{k}$ with
$$\operatorname{ker}_{H_{\beta^{\prime}}^{k}}(B)=Z_{\beta^{\prime}}^{k} \oplus \delta^{*}\left(\operatorname{ker}_{H_{\beta^{\prime}+1}^{k+1}}\left(\Delta_{1}\right)\right)$$
which additionally satisfies
$$Z_{\beta^{\prime}}^{k} \subset Z_{\beta}^{k} .$$

Due to (4.8), we can write
$$h_{1}=\delta^{*} \omega_{2}+h_{2}+k_{2}$$
with $\omega_{2} \in H_{\beta^{\prime}+1}^{k+1}$ and $k_{2}$ is a tensor with support in a small region. Rearranging yields
$$h_{0}=\delta^{*}\left(\omega_{1}+\omega_{2}\right)+h_{2}+k_{2} \subset \delta^{*}\left(H_{\beta}^{k}\left(T^{*} M\right)\right) \oplus Z_{\beta^{\prime}}^{k} \oplus \tilde{V} \subset \delta^{*}\left(H_{\beta}^{k}\left(T^{*} M\right)\right) \oplus Z_{\beta}^{k} \oplus \tilde{V},$$
where we also have taken into account that the sums on the right-hand side are all direct by (4.8). But because $h_{0} \in Z_{\beta}^{k}$, this actually implies that $\delta^{*}\left(\omega_{1}+\omega_{2}\right)=0$. Thus,
$$h_{0}=h_{2}+k_{2} \in Z_{\beta^{\prime}}^{k} \oplus \tilde{V} \subset H_{\beta^{\prime}}^{k}\left(S^{2} M\right)$$
therefore
$$h_{0}=\mathcal{O}_{2}\left(r^{-\xi_{-}^{B}-\epsilon^{\prime}}\right)$$
for some $\epsilon^{\prime}>0$. Pulling back to the cone $\bar{M}$, this implies (by using elliptic regularity) that
$$\varphi_{*} g-\bar{g}=\varphi_{*}\left(g-g_{R}\right)=\mathcal{O}_{\infty}\left(r^{-\xi_{-}^{B}-\epsilon^{\prime}}\right) .$$
By successively improving the decay rate as in the proof of Theorem 4.17 and repeating the above procedure a finite number of times, we obtain
$$\varphi_{*} g-\bar{g}=\varphi_{*}\left(g-g_{R}\right)=\mathcal{O}_{\infty}\left(r^{-\xi_{-}}\right),$$
as desired. \(\square\)

$$\begin{aligned}
& \xi_{+}:=\min E_{+}=\min \left\{\xi_{+}\left(\kappa_{i}\right), \xi_{+}\left(\lambda_{i}\right) \mid i \in \mathbb{N}\right\} \geq 2 \\
& \xi_{-}:=\min E_{-}=\min \left\{-\xi_{-}\left(\kappa_{i}\right),-\xi_{-}\left(\lambda_{i}\right) \mid i \in \mathbb{N}\right\} \geq n .
\end{aligned}$$
The results now follow from Theorem 1.7 \(\square\)
