\section{Verified Results (This Step)}

\subsection{Theorem A (Strengthened Hitchin--Thorpe)}
Any closed 4-manifold admitting $\|\mathrm{Ric} - \Lambda g\|_{\mathrm{op}} \leq \delta|\Lambda|$
for $\Lambda \neq 0$ and $\delta < (\sqrt{3}-1)/2$ satisfies $2\chi > 3|\tau|$.
Eliminates HT-violators and boundary cases from tight pinching.
\emph{Verified}: SymPy + independent TTCS check.

\subsection{Theorem A'' (Corrected Pinching Bound via Prop 4)}
For $M = X \# k\overline{\mathbb{CP}}^2$ with monopole class, under $\|\mathrm{Ric}+g\|_{\mathrm{op}} \leq \delta$:
\[
\frac{k}{A} \leq \frac{1}{3} + \frac{2\delta^2}{(1-\delta)^2},
\qquad A = c_1^2(X).
\]
\emph{Key source}: Ishida--LeBrun Prop 4 gives $\int|W^+|^2 \geq (4\pi^2/3)(a^+)^2$
from $\beta$-independent Cauchy--Schwarz on Weitzenb\"ock for self-dual forms.
\emph{Cross-verified}: S0910A01 Step 3 independent derivation; S0910A02 Step 3 correction confirms.

\textbf{Uniform gap}: All LeBrun manifolds with $k \geq 25A/57$ (no Einstein by Thm B) have
$\delta \geq 1/(1+\sqrt{19}) \approx 0.187$, independent of $A$.

\subsection{Proposition C (DRic Symbol Kernel)}
$\ker\sigma_\xi(D\mathrm{Ric})$ at flat metrics is exactly the gauge space
$\{\xi\otimes v + v\otimes\xi\}$, dimension $n$. Perturbative Ricci homogenization is blocked.
\emph{Verified}: Numerical for $n=3,4,5$.

\subsection{K\"ahler Blow-Up Obstruction (from S0910A01)}
On $\widetilde{X} = X\#k\overline{\mathbb{CP}}^2$ with blow-up complex structure,
ANY K\"ahler metric has $\delta_- \geq 1$. Proof: $c_1(T\widetilde{X})\cdot E_j = 1 > 0$,
but $\mathrm{Ric} \leq -(1-\delta)\omega$ with $\delta < 1$ forces 
$\int_{E_j}\rho \leq -(1-\delta)\mathrm{Area}(E_j) < 0$. Contradiction.
\emph{Consequence}: Any counterexample construction MUST use non-K\"ahler metrics.


\section{Landscape of the Problem (Dimension 4)}

\subsection{Negative branch $P_4^-$}

\textbf{All known no-Einstein 4-manifolds have a positive pinching gap:}
\begin{itemize}
\item HT violators: $\delta \geq (\sqrt{3}-1)/2 \approx 0.366$
\item LeBrun $k/A \geq 2/3$: $\delta \geq 1/(1+\sqrt{6}) \approx 0.290$
\item LeBrun $25/57 \leq k/A < 2/3$: $\delta \geq 1/(1+\sqrt{19}) \approx 0.187$
\item LeBrun $1/3 < k/A < 25/57$: $\delta > 0$ but Einstein not excluded
\item Simplicial volume/entropy obstructions: require infinite $\pi_1$, untested for pinching
\end{itemize}

\textbf{Construction mechanisms all fail:}
\begin{itemize}
\item K\"ahler blow-up: killed by cohomological obstruction ($c_1 \cdot E_j = 1$)
\item ALE gluing with Burns cap: killed by nonzero first obstruction
\item Perturbative homogenization: killed by DRic symbol (pure gauge kernel)
\item Lohkamp $h$-principle: one-sided only ($\mathrm{Ric} < 0$, not two-sided)
\item ARTT convex integration: scalar curvature only; Ricci oscillates
\end{itemize}

\subsection{Positive branch $P_4^+$}

Under tight positive pinching $\|\mathrm{Ric}-g\| \leq \delta$:
\begin{itemize}
\item $s > 0$, killing all SW/monopole obstructions
\item $\pi_1$ finite (Myers), $\mathrm{diam} \leq C/\sqrt{1-\delta}$
\item $2\chi > 3|\tau|$ (strengthened HT)
\item No known 4-manifold has tight positive pinching without Einstein metric
\item Collapse analysis (S0910A02): Hopf collapse has $\delta_+ \to 1$;
      only Einstein quotient collapse (lens spaces) is compatible with $\delta_+ \to 0$
\end{itemize}


\section{Open Questions and Next Directions}

\subsection{For proving $Q_4$ true}

$2\chi > 3|\tau|$, finite $\pi_1$, vanishing SW, and tight positive Ricci pinching
admits an Einstein metric. The main obstacles are:
\begin{enumerate}
\item Weyl curvature control (not given by Ricci pinching alone)
\item Noncollapsing / injectivity radius bounds
\item Ricci flow convergence without sectional curvature bounds
\end{enumerate}

\textbf{Negative branch existence}: Show that tight negative Ricci pinching on a
closed 4-manifold forces Einstein existence. Currently:
\begin{enumerate}
\item Min-Oo requires extra $d^2\max|K|$ bound (not pure Ricci)
\item Anderson convergence uses volume + curvature bounds
\item Ye's modification of Min-Oo: exact theorem still under investigation
\end{enumerate}

\textbf{Candidate approach}: Use the quantitative Chern--Weil structure
(Theorem A'' + Theorem B + Kähler obstruction) to show that tight negative
pinching constrains topology so severely that Einstein existence follows
from the classification of the surviving manifold types.
